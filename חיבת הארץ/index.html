מקורות האהבה במקרא ובמדרש

ארץ ששקולה כנגד כל הבריות

דִּמְיוּ לְרֶגַע מַאֲזְנַיִם עֲנָקִיִּים, שֶׁגְּדוֹלִים דַּיָּם לְהָכִיל אֶת הָעוֹלָם כֻּלּוֹ. בְּכַף אַחַת מֻנָּח כָּל הַיְּצוּר – כָּל אֶרֶץ וְאֶרֶץ, כָּל יַבָּשָׁה וָיָם, כָּל הָר וָעֵמֶק, כָּל כּוֹכָב בַּשָּׁמַיִם, כָּל חַי וְצוֹמֵחַ שֶׁבָּרָא הַקָּדוֹשׁ בָּרוּךְ הוּא בְּשֵׁשֶׁת יְמֵי בְּרֵאשִׁית. תְּמוּנָה עֲצוּמָה שֶׁל שֶׁפַע וָרֹחַב.

וּבַכַּף הַשְּׁנִיָּה – רַק זֹאת: רְצוּעַת אֲדָמָה קְטַנָּה בֵּין הַיַּרְדֵּן לַיָּם. פִּסָּה זְעִירָה בְּתוֹךְ מַפַּת עוֹלָם עֲצוּמָה, שֶׁאֶפְשָׁר לְכַסּוֹת אוֹתָהּ בְּכַף יָד. מִי, לְדַעְתְּכֶם, תַּכְרִיעַ אֶת הַמֹּאזְנַיִם?

חֲכָמֵינוּ עָנוּ בְּלִי הִסּוּס, וּבְלִי לְעַגֵּל פִּנּוֹת. כָּתוּב בְּיַלְקוּט שִׁמְעוֹנִי, בְּפָרָשַׁת עֵקֶב: "רְאֵה מַה חִבָּה חִבֵּב הַקָּדוֹשׁ בָּרוּךְ הוּא אֶת אֶרֶץ יִשְׂרָאֵל, שֶׁשְּׁקוּלָה כְּנֶגֶד כָּל הַבְּרִיּוֹת שֶׁבָּרָא בְּעוֹלָמוֹ". לֹא "כִּמְעַט שָׁקוּל". לֹא "רַב עֵרֶךְ יַחֲסִית לְגָדְלָהּ". שָׁקוּל מַמָּשׁ, כַּף מוּל כַּף. פִּסַּת הָאֲדָמָה הַזּוֹ, מוּל כָּל הַשְּׁאָר.

וּבַעַל 'טְבוּר הָאָרֶץ' לֹא הִסְתַּפֵּק בְּכָךְ, וְהוֹסִיף שָׁכְבָה נוֹסֶפֶת שֶׁל עֹמֶק: "אֵין לְךָ מִצְוָה גְּדוֹלָה מִזּוֹ, לְחַבֵּב אֶרֶץ אֲשֶׁר הַמֶּלֶךְ מַלְכֵי הַמְּלָכִים הַקָּדוֹשׁ בָּרוּךְ הוּא חָפֵץ בִּיקָרָהּ, וְדוֹרֵשׁ אוֹתָהּ תָּמִיד, וְעֵינֵי ה' אֱלֹקֶיךָ בָּהּ. וּבִזְכוּת זֶה נָבוֹא בִּמְהֵרָה אֶל הַמְּנוּחָה וְאֶל הַנַּחֲלָה, וְיִשְׁלַח לָנוּ גּוֹאֵל מְשִׁיחוֹ בִּמְהֵרָה בְּיָמֵינוּ אָמֵן". דִּמְיוּ מֶלֶךְ עֶלְיוֹן, מוֹשֵׁל בְּכָל תֵּבֵל וּמְלוֹאָהּ, שֶׁעֵינָיו נְעוּצוֹת יוֹמָם וָלַיְלָה, בְּלֹא הֶפְסֵק, בְּפִנָּה אַחַת בְּתוֹךְ מַמְלַכְתּוֹ הָעֲנָקִית. זוֹ אֶרֶץ יִשְׂרָאֵל.

וְכָאן חָבוּי הַסּוֹד הַגָּדוֹל, הַנִּסְתָּר לְעֵין רִאשׁוֹנָה, שֶׁמְּעֻצָּב מִמֶּנּוּ כָּל שְׁאָר הַסֵּפֶר שֶׁלְּפָנֵיכֶם: מִצְוַת יִשׁוּב אֶרֶץ יִשְׂרָאֵל אֵינָהּ כְּמוֹ שׁוּם מִצְוָה אַחֶרֶת בַּתּוֹרָה. אֵין חוֹבָה גְּמוּרָה עַל כָּל יְהוּדִי, בְּכָל דּוֹר וְדוֹר, לַעֲלוֹת אֵלֶיהָ בְּפוֹעַל וּמַמָּשׁ – יֵשׁ מִי שֶׁחַיָּיו מְעַכְּבִים אוֹתוֹ, יֵשׁ מִי שֶׁנּוֹלַד וּמֵת בְּחוּצָה לָאָרֶץ בְּלִי לְעוֹלָם לִרְאוֹתָהּ. אֲבָל יֵשׁ חוֹבָה – עַל כָּל יְהוּדִי, בְּכָל מָקוֹם שֶׁהוּא, גַּם בַּפִּנָּה הַמְּרֻחֶקֶת בְּיוֹתֵר בָּעוֹלָם, גַּם בְּמָקוֹם שֶׁמֵּעוֹלָם לֹא רָאָה בּוֹ הַר אוֹ נַחַל שֶׁל אֶרֶץ יִשְׂרָאֵל – לְחַבֵּב אוֹתָהּ בְּלִבּוֹ. וּמִי שֶׁחִבָּתָהּ בּוֹעֶרֶת בְּלִבּוֹ בֶּאֱמֶת, אֵיךְ יוּכַל שֶׁלֹּא לְבַקֵּשׁ, בְּסוֹפוֹ שֶׁל דָּבָר, לַעֲלוֹת אֵלֶיהָ? זוֹ תְּמוּנַת הַפְּתִיחָה שֶׁל כָּל הַסִּפּוּרִים שֶׁיָּבוֹאוּ.

ידבקו – כמו איש באשתו

קְחוּ סֵפֶר תּוֹרָה, וְחַפְּשׂוּ בּוֹ מִצְוָה לֶאֱהֹב אֶת הַטַּלִּית. חַפְּשׂוּ מִצְוָה לֶאֱהֹב אֶת הַתְּפִלִּין, שֶׁמְּלַוּוֹת אוֹתָנוּ בְּכָל בֹּקֶר. חַפְּשׂוּ מִצְוָה לֶאֱהֹב סֻכָּה, שֶׁבָּהּ אָנוּ יוֹשְׁבִים שִׁבְעַת יָמִים בְּכָל שָׁנָה, אוֹ אֶת הַלּוּלָב שֶׁאָנוּ נוֹשְׂאִים בְּיָדֵינוּ בְּשִׂמְחָה. לֹא תִּמְצְאוּ. הַתּוֹרָה מְצַוָּה אוֹתָנוּ לְקַיֵּם מִצְווֹת אֵלּוּ בְּדִיּוּק וּבְהִדּוּר – אַךְ אַף פַּעַם אֵינָהּ דּוֹרֶשֶׁת מֵאִתָּנוּ שֶׁ**נֶּאֱהַב** אוֹתָן.

רַק אַחַת. אֶרֶץ יִשְׂרָאֵל.

הַתּוֹרָה קוֹבַעַת בְּפָרָשַׁת מַסְעֵי (בְּמִדְבָּר לו, ז-ט), פַּעֲמַיִם רְצוּפוֹת, בְּעִנְיַן חֲלֻקַּת נַחֲלוֹת בְּנוֹת צְלָפְחָד: "וְלֹא תִסֹּב נַחֲלָה לִבְנֵי יִשְׂרָאֵל מִמַּטֶּה אֶל מַטֶּה, כִּי אִישׁ בְּנַחֲלַת מַטֵּה אֲבֹתָיו יִדְבְּקוּ בְּנֵי יִשְׂרָאֵל". וְשׁוּב, כַּמָּה פְּסוּקִים לְאַחַר מִכֵּן: "וְלֹא תִסֹּב נַחֲלָה מִמַּטֶּה לְמַטֶּה אַחֵר, כִּי אִישׁ בְּנַחֲלָתוֹ יִדְבְּקוּ מַטּוֹת בְּנֵי יִשְׂרָאֵל". מִלָּה אַחַת, חוֹזֶרֶת פַּעֲמַיִם, וְעוֹלָם שָׁלֵם שֶׁל מַשְׁמָעוּת נִפְתָּח בְּתוֹכָהּ. 'יִדְבְּקוּ' – לֹא 'יֵשְׁבוּ', לֹא 'יַחְזִיקוּ', לֹא 'יִשְׁמְרוּ עַל'. דְּבֵקוּת.

וְאוֹתָהּ מִלָּה בְּדִיּוּק, בְּאוֹתָהּ בִּנְיָן, מוֹפִיעָה בַּתּוֹרָה בְּמָקוֹם אֶחָד מֶרְכָּזִי נוֹסָף בִּלְבַד – בְּיֵסוֹד בְּרִיאַת הָעוֹלָם, בְּתֵאוּר הַנִּשּׂוּאִין הָרִאשׁוֹנִים בַּתּוֹלְדוֹת הָאֱנוֹשׁוּת: "עַל כֵּן יַעֲזָב אִישׁ אֶת אָבִיו וְאֶת אִמּוֹ, וְ**דָבַק בְּאִשְׁתּוֹ**, וְהָיוּ לְבָשָׂר אֶחָד". עַם יִשְׂרָאֵל וְאַרְצוֹ – זֶה זוּג נָשׂוּי. דְּבֵקוּת נַפְשִׁית מַמָּשׁ, לֹא רַק בְּעָלוּת מִשְׁפָּטִית עַל שֶׁטַח.

וְעֵדוּת לְכָךְ אֶפְשָׁר לִמְצֹא אֵצֶל הָרַמְבַּ"ם עַצְמוֹ. פִּתְחוּ אֶת סֵפֶר 'מִשְׁנֵה תּוֹרָה', אַרְבָּעָה עָשָׂר כְּרָכִים כְּבֵדִים הַמְּסַכְּמִים אֶת כָּל הַתּוֹרָה שֶׁבְּעַל־פֶּה, אֶלֶף וְאֶלֶף הֲלָכוֹת מְדֻקְדָּקוֹת. חַפְּשׂוּ בָּהֶם מִצְוָה לְנַשֵּׁק דָּבָר כִּלְשֶׁהוּ בָּעוֹלָם. לֹא תִּמְצְאוּ – חוּץ מֵאַחַת. כָּתַב הָרַמְבַּ"ם, בְּהִלְכוֹת מְלָכִים (ה, י): "גְּדוֹלֵי הַחֲכָמִים הָיוּ מְנַשְּׁקִין עַל תְּחוּמֵי אֶרֶץ יִשְׂרָאֵל וּמְנַשְּׁקִין אֲבָנֶיהָ וּמִתְגַּלְגְּלִין בַּעֲפָרָהּ, וְכֵן הוּא אוֹמֵר: כִּי רָצוּ עֲבָדֶיךָ אֶת אֲבָנֶיהָ וְאֶת עֲפָרָהּ יְחֹנֵנוּ".

אֶרֶץ יִשְׂרָאֵל, אָמְרוּ חֲכָמֵינוּ, הִיא כְּמוֹ אִשְׁתּוֹ שֶׁל עַם יִשְׂרָאֵל (עַיֵּן יְשַׁעְיָהוּ סב, ה). וְעַל אִשָּׁה נֶאֱמַר בְּמַסֶּכֶת יְבָמוֹת (סב:): "הָאוֹהֵב אֶת אִשְׁתּוֹ כְּגוּפוֹ". לֹא יְדִידוּת נְעִימָה. לֹא הַעֲרָכָה מְנֻמֶּסֶת. אַהֲבָה שֶׁל גּוּף וָנֶפֶשׁ גַּם יַחַד, כְּזוּ שֶׁאִישׁ אֵינוֹ יָכוֹל לְהַסְבִּיר לְמִי שֶׁלֹּא חָוָה אוֹתָהּ.

דוד המלך: בחרתי הסתופף

דָּוִד הַמֶּלֶךְ – רֶגֶל רְבִיעִית לַמֶּרְכָּבָה הָעֶלְיוֹנָה, מְשׁוֹרֵר תְּהִלִּים, כּוֹבֵשׁ מְדִינוֹת, מְיַסֵּד שׁוֹשֶׁלֶת מְלָכִים שֶׁתֵּמְשֵׁךְ עַד יְמוֹת הַמָּשִׁיחַ. אִישׁ שֶׁכָּל הָעוֹלָם פָּתוּחַ לְפָנָיו: אַרְמוֹנוֹת, אוֹצְרוֹת, שֻׁלְטָן. עוֹמְדִים אָנוּ, אֵפוֹא, לְרֶגַע נָדִיר – לְגַלּוֹת מָה מִתּוֹךְ כָּל זֶה חָשׁוּב לוֹ בֶּאֱמֶת, בְּמַעֲמַקֵּי נִשְׁמָתוֹ.

וְהַתְּשׁוּבָה מַפְתִּיעָה בְּעֻצְמָתָהּ. אָמְרוּ חֲזַ"ל בַּמִּדְרָשׁ (תַּנְחוּמָא, פָּרָשַׁת רְאֵה): "אַף דָּוִד נִתְאַוָּה לָהּ, שֶׁנֶּאֱמַר (תְּהִלִּים פד): בָּחַרְתִּי הִסְתּוֹפֵף בְּבֵית ה'". לֹא הַכֶּתֶר. לֹא הַנִּצָּחוֹן עַל הָאוֹיְבִים. תַּאֲוַת נַפְשׁוֹ, מְעֵבֶר לְכָל דָּבָר אַחֵר, הָיְתָה עֵבֶר אֶחָד: אֶרֶץ יִשְׂרָאֵל.

וְשׁוֹאֶלֶת הַגְּמָרָא בְּעַצְמָהּ: מַהוּ בְּדִיּוּק 'הִסְתּוֹפֵף'? רַבִּי תַּנְחוּם בְּרַבִּי חֲנִילַאי וְרַב חוֹלְקִים בַּתְּשׁוּבָה, וּשְׁתֵּיהֶן נוֹגְעוֹת לְלֵב. חָכָם אֶחָד אוֹמֵר: אָמַר דָּוִד לִפְנֵי הַקָּדוֹשׁ בָּרוּךְ הוּא – רִבּוֹנוֹ שֶׁל עוֹלָם, אֲפִלּוּ יִהְיוּ לִי פַּרְקְלִטִּין וְטַרְקְלִין, אַרְמוֹנוֹת וְטִירוֹת בְּחוּצָה לָאָרֶץ, וְלֹא יִהְיֶה לִי בְּאֶרֶץ יִשְׂרָאֵל אֶלָּא הַסַּף בִּלְבַד – סַף הַבַּיִת, הָאִסְקֻפָּה שֶׁמֵּעֵבֶר לַדֶּלֶת – בָּחַרְתִּי הִסְתּוֹפֵף. וְהַשֵּׁנִי מוֹסִיף עוֹד מַדְרֵגָה: אֲפִלּוּ לֹא יְהֵא לִי לֶאֱכֹל שָׁם אֶלָּא סְפוּג שֶׁל חֲרוּבִים, הַמַּאֲכָל הַדַּל בְּיוֹתֵר שֶׁיֵּשׁ – בָּחַרְתִּי הִסְתּוֹפֵף. הַמִּדְרָשׁ מְסַיֵּם: "הֱוֵי, וָאֶתֵּן לְךָ אֶרֶץ חֶמְדָּה, אֶרֶץ שֶׁחָמְדוּ לָהּ הָאָבוֹת".

אַרְמוֹן מָלֵא נֶגֶד סַף רֵיק. שֻׁלְחָן עָרוּךְ נֶגֶד פְּרוּסַת חָרוּב יְבֵשָׁה. וְדָוִד הַמֶּלֶךְ, בְּלֹא הֵסוּס, בּוֹחֵר בַּמְּעַט שֶׁבְּאֶרֶץ יִשְׂרָאֵל עַל פְּנֵי הַהַרְבֵּה שֶׁבְּכָל מָקוֹם אַחֵר בָּעוֹלָם.

וְעֵדוּת לְתַאֲוָתוֹ הִשְׁאִיר בְּעַצְמוֹ, בְּתוֹךְ הַסֵּפֶר שֶׁחִבֵּר בְּעַצְמוֹ, סֵפֶר תְּהִלִּים – שֶׁבּוֹ נִמְצָא הַמָּקוֹר הַמְפֹרָשׁ בְּיוֹתֵר לְמִצְוַת חִבַּת הָאָרֶץ בְּכָל הַתָּנָ"ךְ: "כִּי רָצוּ עֲבָדֶיךָ אֶת אֲבָנֶיהָ וְאֶת עֲפָרָהּ יְחֹנֵנוּ" (תְּהִלִּים קב, טו). וּמְבָאֵר הָרַמְבַּ"ן: "רָצוּ – וּתְרַצֵּנִי – כֻּלָּם עִנְיַן חֵפֶץ וְחֵשֶׁק בַּדָּבָר". חֵשֶׁק. תְּשׁוּקָה בּוֹעֶרֶת. לֹא הַכָּרַת תּוֹדָה קָרָה, לֹא חוֹבָה מְמֻסֶּדֶת.

וְעוֹד מְבָאֵר רַבֵּנוּ יוֹנָה אֶת הַמִּלָּה 'יְחֹנֵנוּ' עַצְמָהּ: "כָּתוּב (מִשְׁלֵי יד, כא) 'וּמְחוֹנֵן עֲנָיִים אַשְׁרָיו', כְּלוֹמַר שֶׁהָעֲנִיִּים מוֹצְאִין חֵן בְּעֵינָיו וְנוֹתֵן לָהֶם בְּעַיִן יָפָה... שֶׁעֲפַר אֶרֶץ יִשְׂרָאֵל יִמְצָא חֵן בְּעֵינֵיהֶם". גַּם הֶעָפָר הַסְּתָמִי בְּיוֹתֵר, זֶה שֶׁמִּתַּחַת לְרַגְלַיִם, זוֹכֶה לְאוֹתָהּ נְקֻדַּת חֵן שֶׁבָּהּ מַבִּיטִים עַל עָנִי אָהוּב.

כָּתַב בְּסֵפֶר הַחֲרֵדִים: "צָרִיךְ כָּל אִישׁ יִשְׂרָאֵל לְחַבֵּב אֶת אֶרֶץ יִשְׂרָאֵל וְלָבֹא אֵלֶיהָ מֵאַפְסֵי אֶרֶץ בִּתְשׁוּקָה גְּדוֹלָה, כְּבֵן אֶל חֵיק אִמּוֹ". לֹא כְּעוֹבֵד הַמְּבַצֵּעַ הוֹרָאָה טֶכְנִית. כְּבֵן הַנִּמְשָׁךְ, בְּלִי שֶׁיִּתֵּן עַל כָּךְ אֲפִלּוּ אֶת הַדַּעַת, אֶל חֵיק אִמּוֹ.

בנות צלפחד שחיבבו את הארץ

הַמַּחֲנֶה בְּעַרְבוֹת מוֹאָב סוֹעֵר. חֲמֵשׁ אַחָיוֹת עוֹמְדוֹת לִפְנֵי מֹשֶׁה רַבֵּנוּ וְאֶלְעָזָר הַכֹּהֵן וְהַנְּשִׂיאִים וְכָל הָעֵדָה, בְּפֶתַח אֹהֶל מוֹעֵד, וְתוֹבְעוֹת אֶת זְכוּתָן – זְכוּת שֶׁאִישׁ לֹא הֶעֱנִיק לְנָשִׁים לְפָנֵיהֶן בְּתוֹלְדוֹת הָעֵדָה: נַחֲלָה בָּאָרֶץ שֶׁעֲדַיִן לֹא נִכְבְּשָׁה, שֶׁעֲדַיִן קַיֶּמֶת רַק בַּהַבְטָחָה. "לָמָּה יִגָּרַע שֵׁם אָבִינוּ מִתּוֹךְ מִשְׁפַּחְתּוֹ, כִּי אֵין לוֹ בֵּן? תְּנָה לָּנוּ אֲחֻזָּה בְּתוֹךְ אֲחֵי אָבִינוּ".

וְהַתּוֹרָה, בִּמְקוֹם לִמְנוֹת אוֹתָן בְּפַסְקָנוּת, טוֹרַחַת לְפָרֵט אֶת שְׁמוֹתֵיהֶן בְּדִיּוּק – מַחְלָה נֹעָה וְחָגְלָה וּמִלְכָּה וְתִרְצָה. וְלֹא רַק פַּעַם אַחַת. שָׁלֹשׁ פְּעָמִים חוֹזֶרֶת הַתּוֹרָה עַל שְׁמוֹתֵיהֶן, בִּפְרָשׁוֹת שׁוֹנוֹת. שׁוּם בַּת אַחֶרֶת בְּדוֹר הַמִּדְבָּר – דּוֹר שֶׁל שֵׁשׁ מֵאוֹת אֶלֶף אִישׁ וּמִשְׁפְּחוֹתֵיהֶם – לֹא זָכְתָה לְכָבוֹד הַנָּדִיר הַזֶּה.

מָה גָּרַם לְכָבוֹד הַזֶּה? הַתּוֹרָה מְיַחֶסֶת אוֹתָן בְּמִלִּים שֶׁנִּרְאוֹת, לְמַרְאֵית עַיִן, מְיֻתָּרוֹת: "לְמִשְׁפְּחוֹת מְנַשֶּׁה בֶן יוֹסֵף". וְשׁוֹאֵל רַשִׁ"י שְׁאֵלָה שֶׁנִּרְאֵית טֶכְנִית לְגַמְרֵי, אַךְ מַסְתִּירָה בְּתוֹכָהּ תְּשׁוּבָה מַפְתִּיעָה: וַהֲלֹא כְּבָר נֶאֱמַר לְעֵיל 'בֶּן מְנַשֶּׁה', וְיָדוּעַ שֶׁמְּנַשֶּׁה הוּא בֶּן יוֹסֵף – לָמָּה, אֵפוֹא, כּוֹפֶלֶת הַתּוֹרָה אֶת הַיִּחוּס וּמַזְכִּירָה אֶת יוֹסֵף בְּעַצְמוֹ?

אֶלָּא, מְבָאֵר רַשִׁ"י, לוֹמַר לְךָ: יוֹסֵף חִבֵּב אֶת הָאָרֶץ, שֶׁנֶּאֱמַר (בְּרֵאשִׁית נ, כה) "וְהַעֲלִתֶם אֶת עַצְמֹתַי מִזֶּה" – יוֹסֵף, מִשְׁנֶה לְמֶלֶךְ מִצְרַיִם, בַּעַל כֹּחַ וָעֹשֶׁר שֶׁאֵין כָּמוֹהוּ, מַשְׁבִּיעַ אֶת אֶחָיו שֶׁיַּעֲלוּ אֶת עַצְמוֹתָיו מִמִּצְרַיִם לְאֶרֶץ יִשְׂרָאֵל – וּבְנוֹתָיו, דּוֹרוֹת אַחֲרָיו, חִבְּבוּ אֶת הָאָרֶץ. הַחִבָּה עָבְרָה מִדּוֹר לְדוֹר, מֵאָב לְבָנוֹת, כְּמוֹ נַחֲלָה רוּחָנִית שֶׁלֹּא נִגְמֶרֶת. וְנוֹסִיף עוֹד: רַשִׁ"י מְלַמְּדֵנוּ שֶׁמִּי שֶׁהַתּוֹרָה פִּרְטָה בּוֹ שֶׁ"יִּחֲסוֹ לְשֶׁבַח" – הֲרֵי הוּא צַדִּיק בֶּן צַדִּיק. עַל בְּסִיס זֶה נֶחְשְׁבוּ בְּנוֹת צְלָפְחָד לְצִדְקָנִיּוֹת מַמָּשׁ – עַל אַהֲבָה לְאֶרֶץ שֶׁמֵּעוֹלָם לֹא רָאוּ בְּעֵינֵיהֶן, שֶׁעֲדַיִן לֹא נִכְבְּשָׁה, שֶׁהָיְתָה רַק הַבְטָחָה בָּאֹפֶק.

רבי חנינא מתקן את מכשולי העיר

תַּאֲרוּ לְעַצְמְכֶם רְחוֹב עַתִּיק בְּאֶרֶץ יִשְׂרָאֵל, לִפְנֵי כְּאֶלֶף וּשְׁמוֹנֶה מֵאוֹת שָׁנָה. אֵין אַסְפַלְט, אֵין תְּאוּרַת רְחוֹב. יֵשׁ אֲדָמָה חֲרוּשָׁה, אֲבָנִים בּוֹלְטוֹת, בּוֹרוֹת קְטַנִּים שֶׁיְּכוֹלִים לְהַכְשִׁיל הוֹלֵךְ רֶגֶל בְּרֶדֶת הַלַּיְלָה. וְעוֹבֵר שָׁם, יוֹם אַחֲרֵי יוֹם, יְהוּדִי זָקֵן – אֶחָד מִגְּדוֹלֵי חַכְמֵי הַתַּלְמוּד – וְכוֹרֵעַ, וּמְסַלֵּק אֶבֶן, וּמְיַשֵּׁר בּוֹר, וְהוֹלֵךְ הָלְאָה.

שׁוּב וָשׁוּב, יוֹם אַחַר יוֹם. אִישׁ לֹא שָׁלַח אוֹתוֹ. אֵין זֶה תַּפְקִידוֹ הָרִשְׁמִי. וּבְכָל זֹאת. מְסַפֶּרֶת הַגְּמָרָא בִּכְתוּבּוֹת (קיב.): "רַבִּי חֲנִינָא מְתַקֵּן מַתְקְלֵיהּ" – רַבִּי חֲנִינָא הָיָה מְתַקֵּן וּמְיַשֵּׁר אֶת מִכְשׁוֹלֵי הָעִיר. וְלִשְׁאֵלָה הָעוֹלָה מֵאֵלֶיהָ – מַדּוּעַ? – עוֹנֶה רַשִׁ"י תְּשׁוּבָה שֶׁמְּשַׁנָּה אֶת כָּל זָוִית הָרְאִיָּה: "מֵחֲמַת חִבַּת הָאָרֶץ שֶׁהָיְתָה חֲבִיבָה עָלָיו, וּמְחַזֵּר שֶׁלֹּא יֵצֵא שֵׁם רַע עַל הַדְּרָכִים".

שִׂימוּ לֵב לְמָה שֶׁלֹּא כָּתוּב. לֹא כָּתוּב שֶׁחָשַׁשׁ לְבִטְחוֹן הַהוֹלְכִים, וְלֹא שֶׁרָצָה לְמַלֵּא מִצְוַת הֲסָרַת נֵזֶק בִּלְבַד. הַסִּבָּה הָאֲמִתִּית עֲמֻקָּה יוֹתֵר: דְּאָגָה לִכְבוֹד אֶרֶץ יִשְׂרָאֵל עַצְמָהּ. שֶׁלֹּא יֹאמְרוּ עָלֶיהָ, חַס וְשָׁלוֹם, שֶׁדַּרְכֵיהָ מֻזְנָחוֹת.

חִשְׁבוּ עַל כָּךְ בְּעֵינֵי אֵם: אָדָם שֶׁדּוֹאֵג לְכָל פֶּגַע קָטָן בְּבֵית מִשְׁפַּחְתּוֹ הָאֲהוּבָה – מְתַקֵּן כָּל סֶדֶק בְּקִיר, מְנַקֶּה כָּל אָבָק – לֹא מִפְּנֵי שֶׁמִּישֶׁהוּ הוֹרָה לוֹ, אֶלָּא מִפְּנֵי שֶׁזֶּה בֵּיתוֹ, וְהוּא אֵינוֹ יָכוֹל לִרְאוֹת בִּפְגָמִים בּוֹ. כָּךְ בְּדִיּוּק דָּאַג רַבִּי חֲנִינָא, יוֹם אַחַר יוֹם, לְכָל פֶּגַע בְּאַדְמַת אַרְצוֹ.

רבי אמי ורבי אסי – בין שמש לצל

קַיִץ שָׁרָב בְּאֶרֶץ יִשְׂרָאֵל. שְׁנֵי חֲכָמִים זְקֵנִים, רַבִּי אַמִּי וְרַבִּי אַסִּי, יוֹשְׁבִים תַּחַת הַשֶּׁמֶשׁ הַלּוֹהֶטֶת. הַחֹם קָשֶׁה, כִּמְעַט בִּלְתִּי נִסְבָּל. מְצַפִּים אַתֶּם שֶׁיָּקוּמוּ וְיִתְלוֹנְנוּ עַל מֶזֶג הָאֲוִיר, כְּמוֹ כָּל אָדָם אַחֵר? הֵם קָמִים – אֲבָל לֹא לְהִתְלוֹנֵן. הֵם עוֹבְרִים בְּשֶׁקֶט אֶל הַצֵּל.

וּבַחֹרֶף, כְּשֶׁהַקֹּר עוֹשֶׂה אֶת שֶׁלּוֹ, הֵם עוֹשִׂים בְּדִיּוּק אֶת הַהֵפֶךְ – עוֹבְרִים מֵהַצֵּל אֶל הַשֶּׁמֶשׁ. מְסַפֶּרֶת הַגְּמָרָא (כְּתוּבּוֹת קיב, א): "רַבִּי אַמִּי וְרַבִּי אַסִּי הֲווֹ קָיְימֵי מִשִּׁמְשָׁא לְטוּלָא וּמִטּוּלָא לְשִׁמְשָׁא". עַד כָּאן זֶה נִשְׁמָע כְּמוֹ הֶגְיוֹן פָּשׁוּט שֶׁל אֲנָשִׁים מְנֻסִּים – מְחַפְּשִׂים נוֹחוּת. אֲבָל לָמָּה בִּכְלָל טוֹרַחַת הַגְּמָרָא לְסַפֵּר לָנוּ עַל כָּךְ? הֲרֵי זֶה מַה שֶּׁכָּל אָדָם עוֹשֶׂה.

הַתְּשׁוּבָה נִמְצֵאת אֵצֶל הָר"י מִיגָאש, וְהִיא מְהַפֶּכֶת אֶת כָּל הַתְּמוּנָה מִן הַיְּסוֹד. "פֵּרוּשׁ", כּוֹתֵב הָר"י מִיגָאש בִּתְשׁוּבָתוֹ (סִימָן לט), "מֵחֲמַת חֲבִיבוּתָא דְּאַרְעָא דְיִשְׂרָאֵל עֲלַיְיהוּ" – מִתּוֹךְ חִבָּתָם הָעֲצוּמָה לְאֶרֶץ יִשְׂרָאֵל בִּלְבַד, וְלֹא מִשּׁוּם נוֹחוּת סְתָמִית! הֵם לֹא בָּרְחוּ מֵהַחֹם. הֵם חִפְּשׂוּ הֲנָאָה. כְּשֶׁהָיוּ יוֹשְׁבִים בַּשֶּׁמֶשׁ וּמִצְטַעֲרִים מֵחֹם, קָמוּ וְעָבְרוּ לַצֵּל, "כִּי הֵיכִי דְּלִימְטֵי לְהוּ רוּחָא מֵהַהוּא טוּלָא וּמִשְׁכְּחֵי הֲנָאָה מֵאַרְעָא דְיִשְׂרָאֵל" – כְּדֵי לְמַצּוֹת מִן הָאָרֶץ אֶת הַהֲנָאָה שֶׁבָּהּ. וּכְשֶׁהָיוּ יוֹשְׁבִים בַּצֵּל וּמִצְטַנְּנִים, קָמוּ וְעָבְרוּ לַשֶּׁמֶשׁ, "וּמַטְיָא לְהוּ הֲנָאָה מֵאַרְעָא דְיִשְׂרָאֵל, וּמִתְחַדֵּשׁ לְהוּ בָּהּ שִׂמְחָה בְּכֻלְּהוּ עִדָּנֵי".

לֹא הִסְתַּפְּקוּ בִּישִׁיבָתָם בָּאָרֶץ, כְּמִי שֶׁמְּמַלֵּא חוֹבָה. הֵם רָדְפוּ אַחַר כָּל רֶגַע נוֹסָף שֶׁל הֲנָאָה מִמֶּנָּה, קָמוּ וְנָעוּ בְּמֵשֶׁךְ הַיּוֹם כֻּלּוֹ כְּדֵי לִמְצֹא בּוֹ שִׂמְחָה מִתְחַדֶּשֶׁת – בְּדִיּוּק כְּמוֹ אוֹהֵב שֶׁאֵינוֹ מְסַתְּפֵּק בְּעֶצֶם נוֹכְחוּת הָאֲהוּבָה עָלָיו, אֶלָּא רוֹצֶה לְהַרְגִּישׁ אוֹתָהּ בְּכָל רֶגַע וְרֶגַע מֵחָדָשׁ.

טובה הארץ מאד מאד

שְׁנֵים עָשָׂר מְרַגְּלִים חוֹזְרִים מִתּוּר הָאָרֶץ, אַחֲרֵי אַרְבָּעִים יוֹם שֶׁל הֲלִיכָה מֵעֵבֶר לְעֵבֶר. הֶעָם כֻּלּוֹ מְמַתֵּן, מְצַפֶּה. הָעֶשְׂרָה מְדַבְּרִים בְּקוֹל אֶחָד – עַם חָזָק, עָרִים בְּצוּרוֹת עַד הַשָּׁמַיִם, "וְגַם יְלִדֵי הָעֲנָק רָאִינוּ שָׁם", אֵין סִכּוּי. פַּחַד גּוֹלֵשׁ בְּתוֹךְ הַמַּחֲנֶה, בּוֹכִים בַּלַּיְלָה, כִּמְעַט מַהְפֵּכָה שֶׁל יֵאוּשׁ.

וּמוּל כָּל זֶה, קָם אִישׁ אֶחָד. כָּלֵב בֶּן יְפֻנֶּה. הוּא לֹא מִתְפַּלְפֵּל, לֹא מְנַסֶּה לִבְנוֹת טַעֲנָה מְסֻבֶּכֶת נֶגֶד עֶשְׂרָה עֵדִים. הוּא זוֹעֵק שֵׁם אֶחָד, קָצָר, שֶׁמַּכֶּה כְּמוֹ פַּטִּישׁ: "טוֹבָה הָאָרֶץ מְאֹד מְאֹד" (בְּמִדְבָּר יד, ז).

וְכָאן חֲבוּיָה נְקֻדָּה שֶׁקַּל מְאֹד לְפַסֵּחַ עָלֶיהָ בִּקְרִיאָה מְהִירָה. שֶׁבַח כָּפוּל זֶה, שֶׁל "טוֹב מְאֹד מְאֹד" – שְׁתֵּי פְּעָמִים "מְאֹד" רְצוּפוֹת – אֵינוֹ מוֹפִיעַ בַּתּוֹרָה עַל שׁוּם דָּבָר אַחֵר. חַפְּשׂוּ בְּכָל הַתּוֹרָה כֻּלָּהּ. אֲפִלּוּ בְּסִכּוּם בְּרִיאַת הָעוֹלָם, בְּיוֹם הַשִּׁשִּׁי, כְּשֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא מִסְתַּכֵּל עַל כָּל מַה שֶּׁבָּרָא וְרוֹאֶה כִּי טוֹב – שָׁם נֶאֱמַר רַק "וְהִנֵּה טוֹב מְאֹד" (בְּרֵאשִׁית א, לא), בְּלֹא כֶּפֶל.

אֶרֶץ קְטַנָּה אַחַת זוֹכָה לְשֶׁבַח שֶׁכָּל הָעוֹלָם כֻּלּוֹ, עַל כָּל יְפִי בְּרִיאָתוֹ, לֹא זָכָה לוֹ. מִכָּאן לָמְדוּ שֶׁיַּחַס הַתּוֹרָה לְאֶרֶץ יִשְׂרָאֵל הוּא שֶׁל דָּבָר יָחִיד וּמְיֻחָד בְּעוֹלָם, שֶׁצָּרִיךְ לְחַבְּבוֹ עַד מְאֹד – וּכְשֶׁמְּנַסִּים לְהוֹצִיא עָלָיו דִּבָּה, כְּמוֹ שֶׁעָשׂוּ עֲשָׂרָה מְרַגְּלִים בְּפַחְדָּם, לַעֲמֹד כְּמוֹ כָּלֵב, לְבַד מוּל כֻּלָּם אִם צָרִיךְ, וּלְהַכְרִיז: טוֹבָה הָאָרֶץ מְאֹד מְאֹד.

השבועה של רבי ישראל אבא ציטרון

קַיִץ תרע"ד. הָעוֹלָם כֻּלּוֹ נִגְרָר, כִּמְעַט בְּלִי הַתְרָאָה, לְתוֹךְ מִלְחֶמֶת עוֹלָם. גֶּרְמַנְיָה וְתֻרְכִּיָּה מִצַּד אֶחָד, רוּסְיָה וְאַנְגְּלִיָּה מִצַּד שֵׁנִי. וְאֶרֶץ יִשְׂרָאֵל, שֶׁתַּחַת שִׁלְטוֹן תֻּרְכִּיָּה, נִסְחֶפֶת אֶל תּוֹךְ הַסְּעָרָה. הַמֶּמְשָׁלָה הַתֻּרְכִּית מְפַרְסֶמֶת חוֹק אַכְזָרִי בְּתַכְלִית הָאַכְזָרִיּוּת: כָּל מִי שֶׁאֵינוֹ נְתִין תֻּרְכִּי – מֻכְרָח לַעֲזֹב אֶת הָאָרֶץ. לֹא "מֻמְלָץ". לֹא "רָצוּי". מֻכְרָח.

פִּתְאוֹם, אֶלֶף וְתֵשַׁע מֵאוֹת יְהוּדִים בְּאֶרֶץ יִשְׂרָאֵל – רֻבָּם עוֹלִים מֵרוּסְיָה, שֶׁחָיוּ שָׁם שָׁנִים, שֶׁבָּנוּ שָׁם בָּתִּים וּמִשְׁפָּחוֹת – עוֹמְדִים בִּפְנֵי גֵּרוּשׁ. רַבִּים אָכֵן גֹּרְשׁוּ, נִשְׁלְחוּ בָּאֳנִיּוֹת בְּחֹסֶר כֹּל. אֲבָל הָיוּ שֶׁהֶחְלִיטוּ אַחֶרֶת: לְקַבֵּל עֲלֵיהֶם אֶת הַנְּתִינוּת הָעוֹתְמָנִית, עַל כָּל חֻקֶּיהָ הַקָּשִׁים וְהַזָּרִים – וּבִלְבַד שֶׁיִּשָּׁאֲרוּ בְּאַדְמַת אֶרֶץ יִשְׂרָאֵל.

בֵּינֵיהֶם – רַבִּי יִשְׂרָאֵל אַבָּא צִיטְרוֹן זְצ"ל, רַבָּהּ שֶׁל פֶּתַח תִּקְוָה. אָדָם שֶׁכְּבָר קֹדֶם לָכֵן דָּחָה הַזְמָנוֹת רַבּוֹת לְכַהֵן כְּרַב בִּקְהִלּוֹת מְכֻבָּדוֹת בְּחוּצָה לָאָרֶץ, בְּאָמְרוֹ: "מוּטָב לִהְיוֹת רַב בְּכְפָר קָטָן בְּאֶרֶץ יִשְׂרָאֵל, מֵאֲשֶׁר בְּכֶרֶךְ גָּדוֹל בְּחוּצָה לָאָרֶץ". וְעַכְשָׁו, אָדָם שֶׁכָּל חַיָּיו בָּחַר בָּאָרֶץ, עוֹמֵד לְאַבֵּד אוֹתָהּ בְּעֶצֶם דֶּקֶר שֶׁל חֹק.

הוּא הוֹלֵךְ לְתַחֲנַת הַמִּשְׁטָרָה הַמְּקוֹמִית, מַצִּיעַ לִפְנֵי הַשּׁוֹטְרִים אֶת רְצוֹנוֹ לְקַבֵּל אֶת הַנְּתִינוּת. וְשָׁם מְמַתֶּנֶת לוֹ הַפְתָּעָה קָשָׁה: הַנְּתִינוּת נִתֶּנֶת רַק לְמִי שֶׁנּוֹלַד בָּאָרֶץ. וְעָלָיו לְהִשָּׁבַע עַל כָּךְ בִּפְנֵי הַשּׁוֹטְרִים, אוֹ לְפָחוֹת לְהַצְהִיר בְּהֵן צֶדֶק שֶׁלּוֹ שֶׁהוּא נְתִין הָאָרֶץ.

הוּא לֹא נוֹלַד בָּאָרֶץ. הוּא עוֹזֵב אֶת הַמִּשְׁטָרָה נָבוֹךְ, בֵּין שְׁתֵּי אֶפְשָׁרֻיּוֹת קָשׁוֹת: לְהִשָּׁבַע שְׁבוּעָה שֶׁאֵינָהּ אֱמֶת, אוֹ לַעֲזֹב אֶת אַרְצוֹ הָאֲהוּבָה. הוּא לֹא יוֹדֵעַ מָה לְהַחְלִיט. עוֹלֶה בְּדַעְתּוֹ רַעְיוֹן: לְהִתְיָעֵץ עִם חָמִיו, הַגָּאוֹן רַבִּי יוֹסֵף רוֹזִין מִדְּוִינְסְק, בַּעַל ה'צְפְנַת פַּעֲנֵחַ' – אֶחָד מִגְּאוֹנֵי הַדּוֹר, אִישׁ שֶׁתְּשׁוּבוֹתָיו נֶחְשְׁבוֹת כִּמְעַט לְהֶכְרֵעוֹת שֶׁל שָׁמַיִם.

בְּאוֹתוֹ יוֹם הוּא כּוֹתֵב מִכְתָּב, וּמְתָאֵר אֶת דְּרִישַׁת הַמִּשְׁטָרָה הַתֻּרְכִּית, "וְכִי צַר לִי מְאֹד לַעֲזֹב אֶת אַרְצֵנוּ הַקְּדוֹשָׁה, אֲבָל אִם אֶהְיֶה נְתִין תֻּרְכִּי אוּכַל לְהִשָּׁאֵר, וּבְתַנַאי שֶׁאֹמַר בְּהֵן צֶדֶק שֶׁלִּי שֶׁנּוֹלַדְנוּ בְּאֶרֶץ יִשְׂרָאֵל, וַהֲרֵינִי שׁוֹאֵל אֶת חַוַּת דַּעְתּוֹ הַגְּדוֹלָה בְּנִדּוֹן זֶה". וְשׁוֹלֵחַ אֶת הַמִּכְתָּב.

וְאָז – הַמְּתָנָה. שְׁבוּעַיִם שְׁלֵמִים שֶׁל צִפִּיָּה, כְּשֶׁגּוֹרָלוֹ תָּלוּי בְּתוֹךְ הַסָּפֵק. וּבְסוֹפָם, מַגִּיעָה גְּלוּיָה. לֹא מִכְתָּב אָרֹךְ עִם נִמּוּקִים מְפֻלְפָּלִים. כַּמָּה שׁוּרוֹת בִּלְבַד, וּבָהֶן פְּסַק דִּין הִיסְטוֹרִי: "בְּוַדַּאי מֻתָּר, שֶׁכֵּן נֶאֱמַר: וּלְצִיּוֹן יֵאָמֵר אִישׁ וָאִישׁ יֻלַּד בָּהּ – אֶחָד הַנּוֹלָד בָּהּ וְאֶחָד הַמְצַפֶּה לִרְאוֹתָהּ". רַבִּי יִשְׂרָאֵל אַבָּא, כְּעֵדוּתוֹ הוּא עַצְמוֹ, שָׁמַע בְּקוֹלוֹ, הִצְהִיר שֶׁנּוֹלַד בְּאֶרֶץ יִשְׂרָאֵל, וְנִהְיָה לְנָתִין תֻּרְכִּי – וְנוֹתַר בְּתוֹךְ עַמּוֹ בְּאֶרֶץ הַקֹּדֶשׁ.

וְעַל כָּךְ הוֹסִיף בְּעַל 'בְּאֵר מֹשֶׁה' תוֹבָנָה מַפְתִּיעָה, שֶׁהוֹפֶכֶת סִפּוּר קָטָן זֶה לִיסוֹד גָּדוֹל: מִכָּאן הוֹכָחָה שֶׁאֵין עָלֶיהָ תְּשׁוּבָה, שֶׁאֶרֶץ יִשְׂרָאֵל מְקֻדֶּשֶׁת מִכָּל הָאֲרָצוֹת. שֶׁכֵּן בְּכָל אֶרֶץ אַחֶרֶת, אָדָם הַיּוֹשֵׁב בְּאֶרֶץ אַחֶרֶת וּמְצַפֶּה לִרְאוֹת אֶת אַרְצוֹ – הַאִם יֵחָשֵׁב כְּאִלּוּ נוֹלַד בָּהּ? לֹא. אֲבָל לֹא כֵּן אֶרֶץ יִשְׂרָאֵל, שֶׁכֻּלָּהּ רוּחָנִיּוּת, וּלְפִיכָךְ אַף הַמְצַפֶּה לִרְאוֹתָהּ מַעֲלִים עָלָיו כְּאִלּוּ נוֹלַד בָּהּ – מִשּׁוּם שֶׁהַצְּפִיָּה עַצְמָהּ תּוֹעֶלֶת רוּחָנִית מְרֻבָּה יֵשׁ בָּהּ. מִכָּאן נִלְמָד גַּם אֲנַחְנוּ, הַיּוֹשְׁבִים בָּהּ הַיּוֹם: כְּדֵי לִזְכּוֹת לְמַעֲלוֹתֶיהָ, צָרִיךְ לְהִשְׁתּוֹקֵק אֵלֶיהָ וּלְחַבְּבָהּ.

הגר"א ותלמידיו — עוד הפעם עולים

מפת גבולות הארץ של הגר"א

אָמְרוּ עָלָיו דָּבָר מַפְתִּיעַ, שֶׁכִּמְעַט קָשֶׁה לְהַאֲמִין בּוֹ: אִם יִתֵּן אִישׁ אֶת כָּל הוֹן בֵּיתוֹ בְּאַהֲבָה שֶׁאָהַב הַגְּרַ"א – הַגָּאוֹן מִוִּילְנָה, גֵּאוֹן שֶׁתּוֹרָתוֹ מְמַלֵּאת עֶשְׂרוֹת סְפָרִים – אֶת אֶרֶץ יִשְׂרָאֵל, בּוֹז יָבוּזוּ לוֹ. כְּלוֹמַר: אֵין לְשַׁעֵר אֶת גֹּדֶל הָאַהֲבָה הַזּוֹ בְּמֻשְׂגֵּי כֶּסֶף אוֹ מַאֲמָץ אֱנוֹשִׁי רָגִיל.

וְהָאַהֲבָה הַזּוֹ לֹא הִתְחִילָה בְּבַגְרוּתוֹ. מְסֻפָּר כִּי כְּבָר בְּיַלְדוּתוֹ, יֶלֶד רַךְ שֶׁכְּבָר נֶחְשַׁב לְגָאוֹן, אָסַף כְּסָפִים – פְּרוּטוֹת קְטַנּוֹת, אוּלַי מַתָּנוֹת שֶׁקִּבֵּל – וְשָׁלַח אוֹתָן לְקוּפַּת 'רַבִּי מֵאִיר בַּעַל הַנֵּס' לִתְמִיכָה בְּיוֹשְׁבֵי אֶרֶץ יִשְׂרָאֵל, אֶרֶץ שֶׁמֵּעוֹלָם לֹא רָאָה בְּעֵינָיו.

שָׁנִים עוֹבְרוֹת, וְהוּא הוֹפֵךְ לְגָאוֹן הַדּוֹר. תַּלְמִידָיו יוֹשְׁבִים לְרַגְלָיו, שׁוֹתִים בְּצָמָא כָּל מִלָּה מִפִּיו – וּבְכָל שִׁעוּר וְשִׁעוּר, בְּלִי קֶשֶׁר לַנּוֹשָׂא הַנִּלְמָד, הוּא חוֹזֵר וְדוֹבֵר עַל חוֹבַת הַיְּשִׁיבָה בָּאָרֶץ, עַל בְּנִיָּתָהּ וְהָקָמַת הֲרִיסוֹתֶיהָ. וְיוֹם אֶחָד הוּא מְבַקֵּשׁ מֵהֶם דָּבָר מוּזָר, כִּמְעַט מְשַׁגֵּעַ: לְהָכִין בִּשְׁבִילוֹ **מַפָּה שֶׁל גְּבוּלוֹת הָאָרֶץ**. אָדָם שֶׁמֵּעוֹלָם לֹא הָיָה שָׁם, לוֹמֵד אֶת גְּבוּלוֹתֶיהָ בְּעַל־פֶּה, נָקוֹדָה אַחַר נָקוֹדָה, כְּמוֹ יוֹדֵעַ שֶׁיּוֹם אֶחָד יִצְטָרֵךְ אוֹתָהּ.

וְהוּא הָיָה מְבָאֵר לְתַלְמִידָיו מַדּוּעַ בִּקֵּשׁ מֹשֶׁה רַבֵּנוּ, שֶׁיָּדַע שֶׁלֹּא יִכָּנֵס לָאָרֶץ, בְּכָל זֹאת **לִרְאוֹת** אוֹתָהּ – מִשּׁוּם שֶׁהִשְׁתּוֹקֵק לָדַעַת אֶת כָּל עִנְיָנֶיהָ, "שֶׁגְּבוּלוֹתֶיהָ וְכָל עִנְיָנֶיהָ הֵן עִנְיְנֵי תוֹרָה" – יְדִיעַת גְּבוּלוֹת הָאָרֶץ אֵינָהּ גֵּאוֹגְרַפְיָה סְתָמִית, אֶלָּא חֵלֶק מִגּוּף הַתּוֹרָה עַצְמָהּ.

וּבְעֶרֶב יָמָיו – סוֹף־סוֹף – הוּא יוֹצֵא לַדֶּרֶךְ בְּעַצְמוֹ. אָדָם זָקֵן, גָּאוֹן שֶׁכָּל דּוֹרוֹ תָּלוּי בְּפִיו, יוֹצֵא לְמַסָּע אָרֹךְ וּמְסֻכָּן בַּיָּמִים הָהֵם. מִן הַדֶּרֶךְ הוּא כּוֹתֵב לְאִשְׁתּוֹ וּלְאִמּוֹ הַזְּקֵנָה מִכְתָּב שֶׁכָּל מִלָּה בּוֹ מַרְגִּישָׁה: "וַאֲנִי תְּהִלָּה לָאֵ־ל נוֹסֵעַ לָאָרֶץ הַקְּדוֹשָׁה שֶׁהַכֹּל מְצַפִּים לִרְאוֹתָהּ, חֶמְדַּת כָּל יִשְׂרָאֵל וְחֶמְדַּת הַשֵּׁם יִתְבָּרֵךְ. כָּל הָעֶלְיוֹנִים וְהַתַּחְתּוֹנִים – תְּשׁוּקָתָם אֵלֶיהָ, וַאֲנִי נוֹסֵעַ בְּשָׁלוֹם בָּרוּךְ ה'".

וְכָאן, בְּדִיּוּק בָּרֶגַע שֶׁהַקּוֹרֵא מְצַפֶּה לְסוֹף שָׂמֵחַ, שֶׁל גָּאוֹן זָקֵן הַמַּגִּיעַ סוֹף־סוֹף לִירוּשָׁלַיִם – הַסִּפּוּר מִשְׁתַּבֵּר בְּלֹא הַתְרָאָה. הַגְּרַ"א לֹא מַגִּיעַ. בְּאֶמְצַע הַדֶּרֶךְ הוּא שָׁב לְבֵיתוֹ, וְאִישׁ אֵינוֹ יוֹדֵעַ עַד הַיּוֹם לְאָן בְּדִיּוּק הִגִּיעַ בְּטֶרֶם שָׁב. הַסִּבָּה נִשְׁאֶרֶת סוֹד עָמוּם. יֵשׁ שֶׁהִסְבִּירוּ: נִשְׁמָתוֹ הָיְתָה בִּבְחִינַת מֹשֶׁה רַבֵּנוּ, שֶׁגַּם הוּא לֹא זָכָה לְהִכָּנֵס לָאָרֶץ – אַךְ שָׁלַח אֶת יְהוֹשֻׁעַ בִּמְקוֹמוֹ לְהַשְׁלִים אֶת הַמַּסָּע. הַגְּרַ"א שָׁלַח אֶת תַּלְמִידָיו.

וְאָכֵן, אַהֲבָתוֹ לָאָרֶץ, מְסֻפָּר, הִתְלַהֲטָה בְּשׁוּבוֹ יוֹתֵר מֵאֲשֶׁר לִפְנֵי צֵאתוֹ, וְהוּא הִלְהִיב אֶת תַּלְמִידָיו לֹא לְהִתְמַהְמֵהַּ וְלָצֵאת מִיָּד לְדַרְכָּם לְאֶרֶץ יִשְׂרָאֵל, לְהִתְיַשֵּׁב דַּוְקָא בִּירוּשָׁלַיִם. עֶשֶׂר שָׁנִים אַחֲרֵי פְּטִירָתוֹ, עָלְתָה הַשַּׁיָּרָה הָרִאשׁוֹנָה שֶׁל תַּלְמִידֵי הַגְּרַ"א. אַחֲרֶיהָ עָלְתָה שַׁיָּרָה נוֹסֶפֶת שֶׁל כְּשִׁבְעִים נֶפֶשׁ, וְעוֹד שְׁלִישִׁית. וּבְרֹאשׁ כֻּלָּן – רַבִּי מְנַחֵם מֶנְדֶּל מִשְׁקְלוֹב, שֶׁנִּקְבַּע לְ"רִאשׁוֹן לְעֵדַת הַפְּרוּשִׁים שֶׁעָלָה לְאֶרֶץ יִשְׂרָאֵל", מֵאֲחוֹרֵי הַגְּרַ"א שֶׁלֹּא זָכָה. מֵאֶרֶץ יִשְׂרָאֵל כָּתַב: "הָאָרֶץ מְבַשֶּׂרֶת, הָאָרֶץ מִתְעוֹרֶרֶת... אֱמֶת מַה נֶּהְדָּר יְשִׁיבַת הָאָרֶץ הַטּוֹבָה! אֱמֶת מַה נִּפְלָא חִבַּת אַרְצֵנוּ! גַּם בְּחוּרְבָּנָהּ אֵין דִּמְיוֹנָהּ, בְּשִׁמְמוֹתָהּ אֵין כְּמוֹתָהּ".

כעליית אברהם אבינו

הַבַּעַל שֵׁם טוֹב מְבַקֵּשׁ לַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל. יוֹצֵא לַדֶּרֶךְ. אֲבָל אַף פַּעַם לֹא מַגִּיעַ. הוּא נִפְטָר בְּלִי לִרְאוֹת אֶת אַדְמַת הַקֹּדֶשׁ בְּעֵינָיו. אֲבָל תְּשׁוּקָה שֶׁל צַדִּיק, כְּמוֹ שֶׁנִּרְאֶה שׁוּב וָשׁוּב לְאֹרֶךְ הַסֵּפֶר הַזֶּה, אֵינָהּ נֶעֱלֶמֶת עִם מוֹתוֹ. הִיא עוֹבֶרֶת הָלְאָה, אֶל תַּלְמִידָיו וַחֲסִידָיו, וְחוֹדֶרֶת אֶל נִשְׁמוֹתֵיהֶם כְּמוֹ אֵשׁ שֶׁמְּעַבִּירִים מִנֵּר לְנֵר.

כְּבָר בְּחַיֵּי הַבַּעַל שֵׁם טוֹב עָלוּ חֲסִידִים רַבִּים – רַבּוֹתָיו, חֲבֵרָיו, תַּלְמִידָיו. מֵהֶם שֶׁעָלוּ בְּהַשְׁפָּעָתוֹ וַעֲזָרָתוֹ, כְּגוֹן גִּיסוֹ רַבִּי אַבְרָהָם גֵּרְשׁוֹן מִקִּיטוֹב, רַבִּי נַחְמָן מֵהוֹרוֹדֶנְקָה. מֵהֶם שֶׁעָמְדוּ לַעֲלוֹת, הִתְחִילוּ בְּהֲכָנוֹת וְיָצְאוּ לַדֶּרֶךְ, אַךְ לֹא נִסְתַּיֵּעַ הַדָּבָר בְּיָדָם – כְּמוֹ הַבַּעַל שֵׁם טוֹב עַצְמוֹ, וְתַלְמִידוֹ הַמֻּבְהָק רַבִּי יַעֲקֹב יוֹסֵף הַמַּגִּיד מִפּוֹלְנָא.

אֲבָל בִּסְבִיבוֹת שְׁנַת תקל"ז לָאֶלֶף הַחֲמִישִׁי, קוֹרֶה מַשֶּׁהוּ גָּדוֹל בְּסֵדֶר גֹּדֶל שׁוֹנֶה לְגַמְרֵי. הָעֶלְיוֹת שֶׁקָּדְמוּ לְכָךְ הָיוּ עֲלִיּוֹת יְחִידִים וַחֲבוּרוֹת קְטַנּוֹת. עַכְשָׁו, ה' מֵעִיר אֶת רוּחָם שֶׁל שְׁנֵי מַנְהִיגֵי חֲסִידוּת גְּדוֹלִים, רַבִּי מְנַחֵם מֶנְדֶּל מוִיטֶבְּסְק וְרַבִּי אַבְרָהָם הַכֹּהֵן מִקָּאלִיסְק, לַעֲלוֹת בְּרֹב עָם וּלְכוֹנֵן בְּאֶרֶץ יִשְׂרָאֵל יִשּׁוּב שָׁלֵם שֶׁל חֲסִידִים וְאַנְשֵׁי מַעֲשֶׂה הַהוֹלְכִים בְּתוֹרַת הַבַּעַל שֵׁם טוֹב.

רַבִּי מְנַחֵם מֶנְדֶּל עוֹמֵד בִּפְנֵי בְּחִירָה עֲצוּמָה: לְהַנְהִיג אֶת הַחֲסִידוּת בְּרוּסְיָה וְלִיטָא – עֵמֶק שָׁלֵם שֶׁל מַעֲמָד, כֹּחַ וְהַשְׁפָּעָה עַל אַלְפֵי חֲסִידִים – אוֹ לְוַתֵּר עַל הַכֹּל וְלָלֶכֶת אֶל אֶרֶץ שֶׁלְּשָׁם, כְּלָשׁוֹן הַמְּסַפְּרִים, "נִכְסְפָה נַפְשׁוֹ וְהִתְאַוְּתָה נִשְׁמָתוֹ". הוּא בּוֹחֵר לְוַתֵּר עַל הַכֹּל וְלַעֲלוֹת.

אֵלָיו נִצְטָרֵף רַבִּי יִשְׂרָאֵל מִפּוֹלוֹצְק, דַּרְשָׁן נִלְהָב, וְתְחִלָּה גַּם רַבִּי שְׁנֵאוּר זַלְמָן מִלַּאדִי, בַּעַל ה'תַּנְיָא', שֶׁחָזַר בְּאֶמְצַע הַדֶּרֶךְ – כַּנִּרְאֶה מִשּׁוּם שֶׁלֹּא רָצָה לְהַנִּיחַ אֶת חֲסִידֵי רַיְיסִין וְלִיטָא כְּצֹאן לְלֹא רוֹעֶה. וּמֵהִתְעוֹרְרוּת זוֹ הִתְאַרְגְּנוּ כִּשְׁלֹשׁ מֵאוֹת אֲנָשִׁים שֶׁנָּטְשׁוּ לִצְמִיתוּת אֶת אֶרֶץ הָעַמִּים – חֲמִשָּׁה אָחוּז מִכְּלַל הַיְּהוּדִים שֶׁחָיוּ אָז בְּאֶרֶץ יִשְׂרָאֵל.

אָמַר עָלָיו הָאַדְמוֹ"ר רַבִּי יִשְׂרָאֵל מֵרוֹז'ין: עֲלִיָּתוֹ שֶׁל רַבִּי מְנַחֵם מֶנְדֶּל "הָיְתָה כַּעֲלִיַּת אַבְרָהָם אָבִינוּ, שֶׁסָּלַל אֶת הַדֶּרֶךְ לְבָנָיו שֶׁיִּכָּנְסוּ כַּעֲבֹר כַּמָּה מֵאוֹת שָׁנִים לָאָרֶץ". תְּשׁוּקָה שֶׁל צַדִּיק אֶחָד, שֶׁלֹּא הִגִּיעַ מֵעוֹלָם לָאָרֶץ, סָלְלָה דֶּרֶךְ שֶׁדּוֹרוֹת שְׁלֵמִים אַחֲרָיו הָלְכוּ בָּהּ.

התייר שרצה לעזוב

צָעִיר אֲמֶרִיקָאִי נִכְנָס לְבֵיתוֹ הַצָּנוּעַ שֶׁל הֶחָזוֹן אִישׁ בִּבְנֵי בְּרַק. תַּיָּר, כְּמוֹ רַבִּים אֲחֵרִים שֶׁבָּאִים לְקַבֵּל בְּרָכָה אוֹ עֵצָה. וְכַדַּאי לָדַעַת: יָדוּעַ הָיָה שֶׁהֶחָזוֹן אִישׁ מְעוֹדֵד גַּם אֶת הַתִּיּוּר בָּאָרֶץ, וְאַף כָּתַב בְּאַחַת מֵאִגְּרוֹתָיו (א, קעו) שֶׁ"הַתִּיּוּר בְּאַרְצֵנוּ כְּשֶׁהוּא לְעַצְמוֹ אֵינֶנּוּ מִשּׁוּם חֲשַׁשׁ אִבּוּד מָמוֹן". לֹא כָּל מִי שֶׁבָּא הָיָה בְּעַל כַּוָּנָה מְלֵאָה לְהִשְׁתַּקֵּעַ.

הֶחָזוֹן אִישׁ שׁוֹאֵל אֶת הַצָּעִיר שְׁאֵלָה פְּשׁוּטָה, כִּמְעַט מְיֻתֶּרֶת: הַאִם בָּאתָ לְהִשְׁתַּקֵּעַ בָּאָרֶץ? "לֹא", עוֹנֶה הַצָּעִיר. סְתָם בִּקּוּר. וְאָז הֶחָזוֹן אִישׁ שׁוֹאֵל שְׁאֵלָה שְׁנִיָּה, שֶׁמַּפְתִּיעָה אֶת הַצָּעִיר לְגַמְרֵי: הַאִם מֻתָּר לָצֵאת מֵאֶרֶץ יִשְׂרָאֵל? הַצָּעִיר מִתְבַּלְבֵּל, וּמְנַסֶּה לְהָגֵן עַל עַצְמוֹ בְּמָה שֶׁשָּׁמַע פַּעַם: "שָׁמַעְתִּי שֶׁאִם בָּאִים אַרְצָה בְּכַוָּנָה מֵרֹאשׁ לֹא לְהִשְׁתַּקֵּעַ בָּהּ, מֻתָּר לַעֲזֹב אוֹתָהּ".

וְעַכְשָׁו בָּא הָרֶגַע. הֶחָזוֹן אִישׁ לֹא כּוֹעֵס. תְּגוּבָתוֹ אַכְזָבָה עֲמֻקָּה יוֹתֵר מִכַּעַס. "תִּמָּהוֹן, תִּמָּהוֹן", הוּא אוֹמֵר, "אַתָּה טוֹרֵחַ כְּדֵי לִמְצֹא דֶּרֶךְ אֵיךְ לַעֲזֹב אֶת הָאָרֶץ, בְּעֵת שֶׁאָנוּ מִתְאַמְּצִים כֻּלָּנוּ דַּוְקָא לִמְצֹא דְּרָכִים אֵיךְ לַעֲלוֹת אַרְצָה!" מָה עֲבוּר הַתַּיָּר הָיְתָה שְׁאֵלָה טֶכְנִית לְגִיטִימִית שֶׁל הֶתֵּר, בִּשְׁבִיל הֶחָזוֹן אִישׁ הָיְתָה זוֹ הֲפִיכַת הַסֵּדֶר עַל פִּיו – חֲתִירָה אֶל הַכִּוּוּן הַלֹּא נָכוֹן.

בְּפַעַם אַחֶרֶת בָּאוּ אֵלָיו בַּחוּרֵי יְשִׁיבָה שֶׁעָלוּ מִן הַגּוֹלָה לִלְמֹד תּוֹרָה בָּאָרֶץ. הֵם שׁוֹאֲלִים שְׁאֵלָה הֲלָכָתִית לְגִיטִימִית לְגַמְרֵי: מַה דִּינָם לְעִנְיַן שְׁמִירַת יוֹם טוֹב שֵׁנִי שֶׁל גָּלֻיּוֹת, מֵאַחַר שֶׁדַּעְתָּם לַחֲזֹר בְּסוֹפוֹ שֶׁל דָּבָר לְחוּץ לָאָרֶץ? וְהֶחָזוֹן אִישׁ אֵינוֹ מַשְׁהֶה אֶת תְּשׁוּבָתוֹ אֲפִלּוּ רֶגַע – הוּא מֵשִׁיב "מִיָּנֵיהּ וּבֵיהּ", מִיָּד: "וְכִי זוֹהִי דֵּעָה בִּכְלָל? אֵיזֶה יְהוּדִי יֵשׁ בְּדַעְתּוֹ לַחֲזֹר מִן הָאָרֶץ?" (פְּאֵר הַדּוֹר עַמּוּד מב).

כמה 'הרבה' צריך?

אָדָם אַחְרַאי עַל בַּת יִשְׂרָאֵל מִצְּפוֹן אֵירוֹפָּה נִגַּשׁ פַּעַם אֶל הֶחָזוֹן אִישׁ, מוּדְאָג בֶּאֱמֶת. הוּא בָּדַק, חִפֵּשׂ, וּלְדַעְתּוֹ לֹא מָצָא – בָּאָרֶץ יִשְׂרָאֵל, הוּא חוֹשֵׁב, אֵין הַרְבֵּה בַּחוּרִים מַתְאִימִים בִּשְׁבִילָהּ. אוּלַי כְּדַאי לְשָׁלְחָהּ לְאָמֶרִיקָה, שָׁם הַמַּעֲגָל רָחָב יוֹתֵר, יֵשׁ יוֹתֵר אֶפְשָׁרֻיּוֹת?

חִיֵּךְ הֶחָזוֹן אִישׁ, וְעָנָה תְּשׁוּבָה קְצָרָה שֶׁכָּל כֻּלָּהּ הֶעָרַת עוֹלָם שֶׁלֹּא נִשְׁכֶּחֶת: "מָה צָרִיךְ שֶׁיִּהְיוּ 'הַרְבֵּה' בַּחוּרִים מַתְאִימִים..." (הַגֶּה"צ רַבִּי דֹּב יָפֶה זְצ"ל, לְעַבְדְּךָ בֶּאֱמֶת). לֹא מְנַגֵּד לְעֶצֶם הַחֶשְׁבּוֹן – מְנַגֵּד לַנְּחִיצוּת שֶׁל "הַרְבֵּה" בִּכְלָל, כְּשֶׁמְּדֻבָּר בְּאֶרֶץ יִשְׂרָאֵל.

מרוב אהבה — קנה מגרש בשותפות

כָּל יְמֵי חַיָּיו בְּאֶרֶץ יִשְׂרָאֵל, מֵעֵת עֲלוֹתוֹ וְעַד פְּטִירָתוֹ, הֶחָזוֹן אִישׁ לֹא הָיָה בַּעַל בַּיִת. כָּל דִּירָה שֶׁבָּהּ דָּר – שְׂכוּרָה. אָדָם שֶׁלֹּא הִשְׁקִיעַ פְּרוּטָה בְּקִיר אֶחָד לְעַצְמוֹ. וְעִם זֹאת, קוֹרֶה דָּבָר מַפְתִּיעַ.

מְרַן זְצ"ל קוֹנֶה, בְּשֻׁתָּפוּת עִם עוֹד רַב, חֶלְקַת אֲדָמָה (מִגְרָשׁ) בְּאֶרֶץ יִשְׂרָאֵל בִּבְנֵי בְּרַק, מִשּׁוּם מִצְוַת יִשׁוּב אֶרֶץ יִשְׂרָאֵל בִּלְבַד – לֹא כְּדֵי לִבְנוֹת עָלָיו בַּיִת, וְלֹא כְּדֵי לְהַרְוִיחַ. אַחַר כָּךְ, מִשּׁוּם אֵיזוֹ סִבָּה, נָתַן מְרַן לְשֻׁתָּפוֹ אֶת חֶלְקוֹ – וְקָנָה מִגְרָשׁ אַחֵר, הַפַּעַם בְּשֻׁתָּפוּת עִם גִּיסוֹ, בִּבְנֵי בְּרַק, סָמוּךְ מַמָּשׁ לִישִׁיבַת תִּפְאֶרֶת צִיּוֹן. חֶלְקוֹ שֶׁל מְרַן עָמַד רֵיק, כָּל יְמֵי חַיָּיו, לְעוֹלָם לֹא נִבְנָה עָלָיו דָּבָר.

אָדָם שֶׁלֹּא בָּזְבֵּז שֶׁקֶל עַל דִּירָה מְרֻוַּחַת, מוֹצֵא כֶּסֶף כְּדֵי לִקְנוֹת אֲדָמָה שֶׁהוּא לְעוֹלָם לֹא יִבְנֶה עָלֶיהָ. "וְכִדֵי שֶׁתִּהְיֶה לוֹ חֶלְקַת קַרְקַע בְּאֶרֶץ יִשְׂרָאֵל קָנָה מִגְרָשׁ" – כָּךְ מְתֹאָר הַדָּבָר בְּ'אָרְחוֹת רַבֵּנוּ' לְבַעַל הַ'קְּהִלּוֹת יַעֲקֹב'.

וְעֵדוּת הַגַּר"ח קָנְיֶבְסְקִי זְצ"ל מוֹסִיפָה נִדְבָּךְ: מְרַן הֶחָזוֹן אִישׁ הָיָה מְיַעֵץ וּמְזָרֵז אֲנָשִׁים אֲחֵרִים לִקְנוֹת קַרְקַע בְּאֶרֶץ יִשְׂרָאֵל מִשּׁוּם מִצְוַת יִשׁוּב אֶרֶץ יִשְׂרָאֵל, גַּם אִם אֵין לָהֶם שׁוּם תָּכְנִית לִבְנוֹת עָלֶיהָ בְּקָרוֹב. הַקַּרְקַע עַצְמָהּ, אָמַר, הִיא הַמִּצְוָה.

לנשק אדמתה

עץ הגויאבה של האדמו\"ר מבעלזא

יְרִיּוֹת. חָסִיד אֶחָד, שֶׁגָּר בִּשְׁכוּנַת "יָמִין מֹשֶׁה" בִּירוּשָׁלַיִם, נֶאֱלָץ לִבְרֹחַ מִבֵּיתוֹ – צַלָּפִים יְרַדְּנִים יוֹרִים לְעֵבֶר הַשְּׁכוּנָה בְּלִי הֶרֶף, מֵעֵבֶר הַשֵּׁנִי שֶׁל הַגְּבוּל. הוּא מְבַקֵּשׁ לִמְכֹּר אֶת דִּירָתוֹ, אֲבָל בַּעַל הַדִּירָה מְעַכֵּב – וְאַף עַל פִּי שֶׁפְּסַק דִּין שֶׁל בֵּית הַדִּין שֶׁל 'הָעֵדָה הַחֲרֵדִית' מְחַיֵּב אוֹתוֹ לְאַפְשֵׁר זֹאת, הוּא עוֹמֵד בְּמֵרוֹ וְעַקְשָׁנוּתוֹ, וְלֹא נוֹתֵן לְמָכְרָהּ. הֶחָסִיד נִקְלָע לְמַצּוּקָה קָשָׁה.

וְעֵת כָּזֹאת, שֶׁל דְּאָגָה קַשָּׁה עַל דִּירָה וּפַרְנָסָה, מִי חוֹשֵׁב עַל שֻׁלְחָן הַצַּדִּיק שֶׁל רֹאשׁ הַשָּׁנָה? הוּא, דַּוְקָא. בְּאוֹתָהּ עֵת שָׁמַר הֶחָסִיד עַל דִּירָתוֹ הָרֵיקָה שֶׁל הָאַדְמוֹ"ר מִבֶּעלְזָא בְּתֵל אָבִיב – מִנְהָג קָבוּעַ שֶׁל הָרַבִּי, שֶׁכָּל שָׁנָה, מֵחַג הַשָּׁבוּעוֹת עַד אַחֲרֵי הַיָּמִים הַנּוֹרָאִים, עוֹבֵר לְקָטָמוֹן שֶׁבִּירוּשָׁלַיִם, וּמַקְפִּיד לְהַפְקִיד שׁוֹמְרִים בַּדִּירָה שֶׁיָּדוּרוּ בָּהּ וְיִשְׁמְרוּ עָלֶיהָ – "מִטְּעָמִים הַכְּמוּסִים עִמּוֹ". וּבְחָצֵר אוֹתָהּ דִּירָה – עֵץ גְּוָיָבָה.

בְּהִתְקָרֵב רֹאשׁ הַשָּׁנָה, עוֹלֶה בְּמַחְשַׁבְתּוֹ שֶׁל הֶחָסִיד רַעְיוֹן: לִקְטֹף מִפֵּרוֹת הָעֵץ וּלְהַעֲלוֹתָם לְשֻׁלְחָנוֹ הַטָּהוֹר שֶׁל הָרַבִּי, לְ"סְעֻדַּת לְדָוִד מִזְמוֹר טִישׁ" בְּלֵיל ב' שֶׁל רֹאשׁ הַשָּׁנָה – סְעֻדָּה שֶׁבָּהּ מְחַלְּקִים פֵּרוֹת הָאִילָן וּפֵרוֹת הָאָרֶץ, קֹדֶם שֶׁנּוֹטְלִים אֶת הַיָּדַיִם. מִמַּחְשָׁבָה לְמַעֲשֶׂה: הוּא קוֹטֵף, מַעֲלֶה אֶת הַפֵּרוֹת לִירוּשָׁלַיִם.

בְּעֶרֶב יוֹם טוֹב, נִכְנָסִים כֻּלָּם – "אִישׁ אִישׁ וּמִנְחַת פֵּרוֹתָיו" – לְהַעֲלוֹת עַל הַשֻּׁלְחָן. מָרָן נִרְאָה כְּמִתְנַמְנֵם, כְּפִי שֶׁהָיָה נִרְאֶה בְּכָל עֵת שֶׁהִזְכִּירוּ לְפָנָיו 'קְוִיטְלֶעךְ'. וְהַמְּשַׁמֵּשׁ, הָרַב שָׁלוֹם פוֹיגֶל, מְנַעְנֵעַ קְמֵעָא אֶת כִּסֵּא קָדְשׁוֹ שֶׁל מָרָן לְהָעִירוֹ. וְאָז מַגִּיעַ הֶחָסִיד עִם טְנָא הַגְּוָיָבוֹת, וְהַמְּשַׁמֵּשׁ רוֹכֵן וְאוֹמֵר: "הוּא הֵבִיא מִפֵּרוֹת הָעֵצִים הַנְּטוּעִים בַּחֲצַר הָרַבִּי שֶׁבְּתֵל אָבִיב".

וְאָז – רֶגַע קָטָן שֶׁמְּגַלֶּה עוֹלָם שָׁלֵם. מָרָן מֵרִים רֹאשׁוֹ בְּאַחַת. עֵינָיו נִפְקָחוֹת. "הַפֵּרוֹת הַלָּלוּ – מֵחֲצֵרִי הֵם? הַאִם הָעֵץ 'אִילָן רַבְרְבָא' הוּא, גָּדוֹל וְנָאֶה? הַאִם הַנְּךָ מַשְׁקֵהוּ וּמְטַפֵּחַ אוֹתוֹ?" שִׂמְחָה וַהֲנָאָה מְרֻבָּה נִשְׁתַּקְּפוּ מִפָּנָיו הַקְּדוֹשׁוֹת בְּדַבְּרוֹ, וְנִרְאֶה וּמוּבָן הָיָה שֶׁשִּׂמְחָה זוֹ בָּאָה לוֹ עַל פֵּרוֹת אֶרֶץ יִשְׂרָאֵל – לֹא עַל שׁוּם דָּבָר אַחֵר.

הֶחָסִיד הֵבִין שֶׁזּוֹ שְׁעַת כֹּשֶׁר וְעֵת רָצוֹן. הוּא מַזְכִּיר אֶת קָשְׁיָיו בְּעִנְיְנֵי הַדִּירָה, וּמָרָן בֵּרֵךְ אוֹתוֹ עַל אֲתַר שֶׁיַּצְלִיחַ. וּבְרַכַּת הַצַּדִּיק מִהֲרָה לְהִתְקַיֵּם: מִיָּד אַחֲרֵי יוֹם הַכִּפּוּרִים חִפֵּשׂ בְּעַל הַדִּירָה בְּעַצְמוֹ אֶת הֶחָסִיד, הִתְנַצֵּל מֵעֹמֶק לִבּוֹ, וְהִתְחַנֵּן שֶׁיִּמְחַל לוֹ, כִּי הֲרֵי יְהוּדִים מוֹחֲלִים זֶה לָזֶה בְּיוֹם הַכִּפּוּרִים, וְאַף הִצִּיעַ לְשַׁלֵּם מִכִּיסוֹ סְכוּם גָּדוֹל מְאֹד עֲבוּר הַדִּירָה, וּבִלְבַד שֶׁיִּמְחַל לוֹ. וּבְאוֹתוֹ סְכוּם קָנָה הֶחָסִיד דִּירָה בִּבְנֵי בְּרַק. "וַיְהִי הַדָּבָר לְפֶלֶא", מְסַכֵּם הַסֵּפֶר 'רְשׁוּמִים בְּשִׁמְךָ'. פֵּרוֹת עֵץ גְּוָיָבָה, בְּרַכַּת צַדִּיק, וּבַיִת חָדָשׁ – כֻּלָּם נֶאֱרָגִים בְּסִפּוּר אֶחָד.

בן מלך עם המטאטא

יְהוּדָה הָרוֹשׁ מְנַקֶּה אַשְׁפָּה בִּרְחוֹבוֹת יְרוּשָׁלַיִם, בִּשְׁכוּנַת 'מִשְׁכָּנוֹת'. יוֹם רָגִיל, כְּמוֹ תָּמִיד – עֶגְלַת זֶבֶל, מַטְאֵטֵא, בּוּז שָׁקֵט שֶׁל עוֹבְרֵי הַדֶּרֶךְ שֶׁכִּמְעַט לֹא רוֹאִים אוֹתוֹ. וְאָז נִגַּשׁ אֵלָיו רַבִּי אַרְיֵה לֵוִין, וְאוֹמֵר לוֹ מִשְׁפָּט שֶׁמְּקַפֵּל אֶת יְהוּדָה בַּמָּקוֹם: "אַתָּה יוֹדֵעַ, יְהוּדָה, טוֹב אַתָּה מִמֶּנִּי. אַשְׁרֶיךָ וְטוֹב לְךָ! מְקַנֵּא אֲנִי בְּךָ!"

יְהוּדָה, מְבֻלְבָּל, מְנַסֶּה לְתַקֵּן אֶת מָה שֶׁנִּרְאֶה כְּמוֹ טָעוּת: "מָה אַתָּה שָׂח, רַבִּי אַרְיֵה", הוּא מֵשִׁיב. "אַתָּה חָכָם וְצַדִּיק, עוֹסֵק כָּל הַיּוֹם בְּתוֹרָה וּבְמִצְווֹת, וַאֲנִי – אָדָם פָּשׁוּט מְנַקֶּה אַשְׁפָּה"... אֲבָל רַבִּי אַרְיֵה לֹא נָסוֹג. הוּא מַפְסִיק אֶת דְּבָרָיו וְאוֹמֵר בִּתְקִיפוּת מְאַהֶבֶת:

"אֵינְךָ יוֹדֵעַ אֶת מַעֲלָתְךָ. סָמוּךְ עָלַי, אֲנִי מַבְטִיחֲךָ נֶאֱמָנָה: טוֹב אַתָּה מִמֶּנִּי, אַתָּה מְתַקֵּן הָעוֹלָם וּמְנַקֶּה אֶת חוּצוֹתֶיהָ שֶׁל יְרוּשָׁלַיִם, בְּמוֹ יָדֶיךָ עוֹשֶׂה אַתָּה אֶת עוֹלָמוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא טוֹב יוֹתֵר, נָקִי יוֹתֵר... מְקַנֵּא אֲנִי בְּךָ".

כָּךְ עָמְדוּ וְהִתְוַכְּחוּ, מְסַפֵּר יְהוּדָה – אֲנִי בְּשֶׁלִּי וְהוּא בְּשֶׁלּוֹ. אַךְ מָה אֹמַר לָכֶם: כְּשֶׁפָּנָה רַבִּי אַרְיֵה לָלֶכֶת, וְנוֹתַרְתִּי לְבַדִּי, עִם הַמַּטְאֵטֵא וְעֶגְלַת הַזֶּבֶל, נַעֲשָׂה לִי לְפֶתַע טוֹב עַל הַלֵּב. הִרְגַּשְׁתִּי כְּמוֹ בֶּן מֶלֶךְ. בַּפַּעַם הָרִאשׁוֹנָה בְּחַיַּי... כְּמוֹ בֶּן מֶלֶךְ.

וְהַסִּפּוּר לֹא נִגְמָר כָּאן. שָׁנִים לְאַחַר מִכֵּן, מְקַבֵּל הַסּוֹפֵר שִׂמְחָה רָז מִכְתָּב לְבֵיתוֹ מִיְּהוּדִי מֵרִאשׁוֹן לְצִיּוֹן, שֶׁמְּסַפֵּר: בְּיַלְדוּתוֹ הָיָה לוֹ קָשֶׁה מְאֹד בַּלִּמּוּדִים, וּבְעִקְבוֹת כָּךְ נֶאֱלַץ לַעֲזֹב אֶת לִמּוּדָיו וְהָפַךְ לִהְיוֹת עוֹבֵד נִקָּיוֹן שֶׁל בֵּית הַסֵּפֶר. דָּבָר זֶה גָּרַם לוֹ לְהַרְגִּישׁ עַצְמוֹ כְּכִשָּׁלוֹן, כְּאִלּוּ חַס וְשָׁלוֹם אֵינוֹ שָׁוֶה שׁוּם דָּבָר.

יוֹם אֶחָד הִגִּיעַ לְיָדוֹ סֵפֶר 'צַדִּיק יְסוֹד עוֹלָם', וְקָרָא בּוֹ אֶת הַסִּפּוּר הַזֶּה – שֶׁבּוֹ אוֹמֵר אוֹתוֹ צַדִּיק מֻפְלָא שֶׁהוּא מְקַנֵּא בְּעוֹבֵד הַנִּקָּיוֹן, וְאוֹמֵר עָלָיו שֶׁהוּא טוֹב מִמֶּנּוּ, שֶׁמַּעֲשָׂיו חֲשׁוּבִים וִיקָרִים עַד מְאֹד, שֶׁמְּנַקֶּה הוּא אֶת עוֹלָמוֹ שֶׁל הַקָּדוֹשׁ בָּרוּךְ הוּא וְהוֹפְכוֹ לְטוֹב יוֹתֵר. "וְאָז הִרְגַּשְׁתִּי מַהְפֵּךְ בְּכָל חַיַּי", הוּא כּוֹתֵב. "אֵינִי כִּשָּׁלוֹן שֶׁאֵינוֹ שָׁוֶה שׁוּם דָּבָר, הֲרֵי מָה שֶׁאֲנִי עוֹשֶׂה הוּא דָּבָר גָּדוֹל וְנִפְלָא, תִּקּוּן הָעוֹלָם. הִתְמַלֵּאתִי שִׂמְחָה, חַיַּי הִתְמַלְּאוּ מַשְׁמָעוּת. עַד שֶׁהֶחְלַטְתִּי שֶׁבְּנוֹסָף לַעֲבוֹדָתִי הָרְגִילָה אֵלֵךְ לְהִתְנַדֵּב כִּמְנַקֶּה בַּיְּשִׁיבָה, לְהוֹסִיף וּלְתַקֵּן אֶת הָעוֹלָם. הֲרֵי אֵינֶנִּי כִּשָּׁלוֹן שֶׁלֹּא שָׁוֶה דָּבָר – אֲנִי בֶּן שֶׁל מֶלֶךְ!"

אבנים כיהלומים

עוֹד בְּטֶרֶם עָלָה לְאֶרֶץ יִשְׂרָאֵל, וְעוֹד בְּטֶרֶם הִגִּיעַ לְזִקְנָתוֹ, שׁוֹמֵעַ הָרַב אַבְרָהָם דּוֹב מֵאַבְרִיטְשׁ, בַּעַל ה'בַּת עַיִן', שָׁלִיחַ מֵאֶרֶץ יִשְׂרָאֵל (שְׁלוּחָא דְּרַבָּנָן) דּוֹרֵשׁ בְּהִתְלַהֲבוּת עַל שֶׁבַח הָאָרֶץ, וּמְפָלִיג בְּדִבְרֵי הִתְפַּעֲלוּתוֹ: "מָה אֹמַר וּמָה אֲדַבֵּר, הֲלֹא אֲפִלּוּ הָאֲבָנִים הַפְּשׁוּטוֹת הַמֻּשְׁלָכוֹת בְּחוּצוֹת אֶרֶץ יִשְׂרָאֵל מְאִירוֹת כַּאֲבָנִים טוֹבוֹת וְכַיַּהֲלוֹמִים". הַדִּמּוּי נֶחְרָט בְּלִבּוֹ הָרַב.

שָׁנִים עוֹבְרוֹת. הָרַב הַזָּקֵן מִתְכּוֹנֵן לַנְּסִיעָה הַנִּכְסֶפֶת, "סוֹפֵר אֶת הַיָּמִים וְאֶת הַשָּׁעוֹת לִקְרַאת יוֹם הַנְּסִיעָה הַנִּכְסָף", אֵינוֹ רוֹאֶה וְאֵינוֹ שׁוֹמֵעַ מְאוּמָה רַק אֶרֶץ יִשְׂרָאֵל וְאֶרֶץ יִשְׂרָאֵל. וְלַבְּסוֹף, אַחֲרֵי מְכַשּׁוֹלוֹת עֲצוּמִים – לֹא קַל לִנְטֹשׁ רַבָּנוּת בְּעִיר מְכֻבֶּדֶת וּפַרְנָסָה בְּהַרְחָבָה – עֲלִיָּה.

הָרַב מַגִּיעַ, סוֹף־סוֹף, לְאֶרֶץ יִשְׂרָאֵל. וְהוּא מְחַפֵּשׂ. מַבִּיט בְּכָל אֶבֶן שֶׁבַּדֶּרֶךְ. אֲבָל הָאֲבָנִים – פְּשׁוּטוֹת. אֵין בָּהֶן שׁוּם בֶּרֶק. הוּא פּוֹגֵשׁ אֶת אוֹתוֹ שָׁלִיחַ, שֶׁכְּבָר הָיָה בָּאָרֶץ, וְאוֹמֵר לוֹ, אוּלַי בְּעֶקְצַת אַכְזָבָה: "חִפַּשְׂתִּי אֶת הָאֲבָנִים הַטּוֹבוֹת וְלֹא מָצָאתִי..."

וְהַשָּׁלִיחַ עוֹנֶה לוֹ מִשְׁפָּט אֶחָד שֶׁמְּטַלְטֵל אֶת הָרַב עַד עֻמְקוֹ: "מִי שֶׁעֵינָיו זַכּוֹת וּנְקִיּוֹת – זוֹכֶה וְרוֹאֶה". אֵין בְּעָיָה בָּאֲבָנִים. הַבְּעָיָה בְּעֵינֵי הָרוֹאֶה.

הַמִּשְׁפָּט הַזֶּה לֹא מַרְפֶּה מִמֶּנּוּ. "מֵעוֹדִי לֹא הֻסְכַּנְתִּי לִשְׁמֹעַ אִישׁ מְדַבֵּר אֵלַי בִּתְקִיפוּת כְּפִי שֶׁהַמְּשֻׁלָּח דִּבֵּר", הוּא מוֹדֶה לְתַלְמִידוֹ בְּמְאֻחָר יוֹתֵר. הָרַב מִסְתַּגֵּר בְּחַדְרוֹ – שָׁנָה תְּמִימָה – לְהִזְדַּכֵּךְ וּלְהִתְקַדֵּשׁ. וְכַעֲבֹר אוֹתָהּ שָׁנָה, יוֹצֵא מֵחַדְרוֹ בְּשִׂמְחָה, וְעוֹרֵךְ סְעֻדָּה גְּדוֹלָה. "עַכְשָׁו שֶׁזָּכִיתִי לְכָךְ", הוּא אוֹמֵר, "לִרְאוֹת אֶת הָאֲבָנִים בְּחוּצוֹת כְּיַהֲלוֹמִים וַאֲבָנִים טוֹבוֹת – הִנְנִי עוֹשֶׂה סְעֻדָּה גְּדוֹלָה, לְהוֹדוֹת לַה' יִתְבָּרֵךְ". וְהַכָּתוּב שֶׁבָּחַר לְצַטֵּט מְתָאֵר בְּדִיּוּק אֶת מָה שֶׁחָווָה: "וְשַׂמְתִּי כַּדְכֹד שִׁמְשֹׁתַיִךְ... וְכָל גְּבוּלֵךְ לְאַבְנֵי חֵפֶץ" (יְשַׁעְיָהוּ נד, יב). שָׁנָה שְׁלֵמָה שֶׁל עֲבוֹדָה פְּנִימִית, כְּדֵי לִזְכּוֹת לִרְאוֹת מָה שֶׁתָּמִיד הָיָה שָׁם.

מוּל הַמָּלוֹן בְּצָרְפַת – רַכֶּבֶל הַמּוֹבִיל לְאֶחָד מֵרִכְסֵי הָאַלְפִּים הַמַּרְהִיבִים בְּיוֹתֵר בָּעוֹלָם. כֶּנֶס רַבָּנִים שְׁנָתִי שֶׁזִּמֵּן הָרַב סִיטְרוּק, רַבָּהּ שֶׁל צָרְפַת, לְכָל רַבָּנֵי הַמְּדִינָה, וּבְתוֹךְ הַהַפְסָקוֹת, רַבִּים מִן הַמִּשְׁתַּתְּפִים מְנַצְּלִים אֶת הַשְּׁהוּת בַּמָּקוֹם כְּדֵי לַחֲזוֹת בְּפִלְאֵי הַבְּרִיאָה. אֲבָל הָרַב שִׂמְחָה הַכֹּהֵן קוּק זְצ"ל, רַב הָעִיר רְחוֹבוֹת, בְּשׁוּם אֹפֶן לֹא אָבָה לָלֶכֶת לְשָׁם.

אֶחָד מִגְּדוֹלֵי הַהוֹרָאָה שֶׁבַּכֶּנֶס אָמַר לוֹ שֶׁיִּתָּכֵן שֶׁלְּהִמָּנַע מִלִּרְאוֹת פֶּלֶא בְּרִיאָה כָּזֶה תְּהֵא עַל כָּךְ תְּבִיעָה בַּשָּׁמַיִם. כֵּיוָן שֶׁכָּךְ, נֵעוֹת הָרַב וְקִבֵּל, וְהִסְכִּים לָלֶכֶת לָרַכֶּבֶל. הָלְכוּ בְּיוֹם שִׁשִּׁי – אַךְ לְשִׂמְחָתוֹ הָרַבָּה, הָרַכֶּבֶל לֹא עָבַד בְּאוֹתָהּ שָׁעָה.

כַּאֲשֶׁר נִשְׁאַל לָמָּה כָּל כָּךְ מְסָרֵב לָלֶכֶת לִרְאוֹת אֶת פִּלְאֵי הַמָּקוֹם, הֵשִׁיב הָרַב: "'וּרְאֵה בְטוּב יְרוּשָׁלַיִם'!... הַדָּבָר הַיָּפֶה בְּיוֹתֵר זֶה אֶרֶץ יִשְׂרָאֵל, אֶת זֶה צָרִיךְ לִרְאוֹת, אֵינֶנִּי רוֹצֶה לִפְגֹּם בַּכִּסּוּפִים לְיֹפִי שֶׁל אֶרֶץ יִשְׂרָאֵל עַל יְדֵי הַיֹּפִי הַמְדֻמֶּה וְהַטָּפֵל שֶׁל אֶרֶץ הָעַמִּים..."

בתים יפים רק בארץ ישראל

מַנְהֶטֶן. גּוֹרְדֵי שְׁחָקִים שֶׁתַּיָּרִים מִכָּל הָעוֹלָם בָּאִים כְּדֵי לְהַבִּיט בָּהֶם. הָאַדְמוֹ"ר מִצַּאנְז, בַּעַל ה'שְׁפַע חַיִּים' זי"ע, נוֹסֵעַ לְרוֹפֵא, וְעוֹבְרִים בַּדֶּרֶךְ מוּל אוֹתָם בִּנְיָנִים גְּבוֹהִים, וְהַמְּלַוִּים לוֹ רוֹצִים לְהַרְאוֹת אוֹתָם לָרַבִּי. אֲבָל הָרַבִּי לֹא מַרְאֶה שׁוּם עִנְיָן – וְאָז מַסְבִּיר.

הַסִּבָּה: כְּשֶׁבָּא לָגוּר בְּאָמֶרִיקָה, נָדַר לְעַצְמוֹ שֶׁלֹּא לִרְאוֹת וְלֹא לֵהָנוֹת מֵהַבָּתִּים הַיָּפִים שָׁם. "בָּתִּים יָפִים בִּרְצוֹנִי לִרְאוֹת רַק בְּאֶרֶץ יִשְׂרָאֵל!" וְאָכֵן, בְּכָל פַּעַם שֶׁנָּסְעוּ דֶּרֶךְ מַנְהֶטֶן, לֹא הֵרִים הָרַבִּי אֶת עֵינָיו הַחוּצָה אֶל בִּנְיְנֵי הַסַּחַר שֶׁמִּכָּל הָעוֹלָם בָּאוּ לִרְאוֹתָם, כְּפִי שֶׁמְּסַפֵּר בְּנוֹ הָאַדְמוֹ"ר מִצַּאנְז שֶׁלִּיטָ"א עַל אָבִיו.

הרבי מבויאן ברכבת

הָרַכֶּבֶת חוֹלֶפֶת בִּתְוַאי נוֹף שֶׁל הָרֵי הָאַלְפִּים, בְּדַרְכּוֹ שֶׁל הָאַדְמוֹ"ר מִבּוֹיָאן זְצ"ל בַּחֲזָרָה מִשְּׁוַיְץ לְאֶרֶץ יִשְׂרָאֵל. נוֹף עוֹצֵר נְשִׁימָה, כָּךְ מְסַפֵּר הָרַב זִילְבֵּר מֵאַרְצוֹת הַבְּרִית, שֶׁרִכֵּז אֵלָיו אֶת מַבְּטֵי כָּל הַנּוֹסְעִים בָּרַכֶּבֶת. כֻּלָּם צוֹפִים בַּחַלּוֹן, מְהֻפְּנָטִים. חוּץ מֵאֶחָד. הָרַבִּי מַמְשִׁיךְ לַעֲסֹק בְּשֶׁלּוֹ, בְּסֵפֶר הַמִּשְׁנָיוֹת שֶׁבְּיָדוֹ, כְּאִלּוּ לֹא אֵרַע דָּבָר, וְלֹא גּוֹרֵעַ עַיִן מִן הַסֵּפֶר כָּל הָעֵת.

הַמְּשַׁמֵּשׁ, בִּרְצוֹתוֹ לִגְרֹם לָרַבִּי קוֹרַת רוּחַ, מֵסֵב אֶת תְּשׂוּמַת לִבּוֹ אֶל הַנּוֹף הַנִּפְלָא. אֲבָל הָרַבִּי מַמְשִׁיךְ בְּלִמּוּדוֹ, לֹא מֵגִיב.

וְעַכְשָׁו – הָהֶכְפֵּל שֶׁמְּעוֹרֵר אֶת הַתְּמִיהָה. כְּשֶׁמַּגִּיעִים לְאֶרֶץ יִשְׂרָאֵל, וְהָרַבִּי עוֹלֶה לָרֶכֶב שֶׁהוֹבִילוֹ לְבֵיתוֹ, הוּא לְפֶתַע מַשְׁקִיף דֶּרֶךְ חַלּוֹנוֹת הָרֶכֶב אֶל הַנּוֹף הַנִּרְאֶה – וְלֹא גּוֹרֵעַ עַיִן מִתְּוַאי הַדֶּרֶךְ. עַכְשָׁו הַמְּשַׁמֵּשׁ לֹא יָכוֹל יוֹתֵר לְהִתְאַפֵּק: "רַבִּי, אֵינֶנִּי מֵבִין. הַאִם הַנּוֹף בִּשְׁוַיְץ לֹא הָיָה יָפֶה עֲשֶׂרֶת מוֹנִים וּמַרְשִׁים?!"

וְהַתְּשׁוּבָה, כְּשֶׁהִיא בָּאָה, פּוֹתַחַת חַלּוֹן לְעוֹלָם אַחֵר לְגַמְרֵי: "שְׁתֵּי תְּשׁוּבוֹת לִי בַּדָּבָר", עוֹנֶה הָרַבִּי. "הָאַחַת, וְכִי אֲנִי צָרִיךְ לְהַבִּיט בַּנּוֹף שֶׁל הָרֵי הָאַלְפִּים כְּדֵי לִרְאוֹת אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא?! הֲרֵי 'מִבְּשָׂרִי אֶחֱזֶה אֱלוֹקַ', אֶפְשָׁר לִרְאוֹת אֶת הַקָּדוֹשׁ בָּרוּךְ הוּא גַּם מִתּוֹךְ הִתְבּוֹנְנוּת בְּכַף הַיָּד, בַּוְּרִידִים הַמַּקִּיפִים אֶת הַגּוּף, בָּעֵינַיִם וּבָרַגְלַיִם, וּבְכָל יֶתֶר הָאֵבָרִים שֶׁבָּרָא לָנוּ הָאֱלֹקִים, הַפּוֹעֲלִים כֻּלָּם יַחְדָּו בְּהַרְמוֹנְיָה מֻפְלָאָה וּמַשְׁלִימִים הָאֶחָד אֶת הַשֵּׁנִי. אִם כֵּן, לֹא הָיִיתִי כְּדַאי לְהַפְסִיק מִלִּמּוּדִי וּלְהַבִּיט בַּנּוֹף הַשְּׁוֵיצָרִי.

"וְאִם תִּשְׁאַל אֵפוֹא, מַדּוּעַ הִתְבּוֹנַנְתִּי בַּנּוֹף כַּאֲשֶׁר הִגַּעְנוּ אַרְצָה, הֲרֵי הַתְּשׁוּבָה לְכָךְ מְפֹרֶשֶׁת בַּפָּסוּק 'אֶרֶץ אֲשֶׁר ה'... דֹּרֵשׁ אֹתָהּ תָּמִיד, עֵינֵי ה' אֱלֹקֶיךָ בָּהּ' (דְּבָרִים יא-יב), אֶרֶץ שֶׁהַקָּדוֹשׁ בָּרוּךְ הוּא מֵישִׁיר אֵלֶיהָ אֶת עֵינָיו, שֶׁאֲנִי לֹא אֶסְתַּכֵּל עָלֶיהָ?!"

הֲרֵי לָנוּ מַבָּט אַחֵר עַל כָּל הַחַיִּים.

האבן והחול של הבן איש חי

בֶּן שְׁלוֹשִׁים וְחָמֵשׁ, בְּעֵיצוּמָהּ שֶׁל הַחַמָּה שֶׁל שְׁנַת תרכ"ט, יוֹצֵא הַבֶּן אִישׁ חַי מִבָּבֶל בְּמַסָּע אֶל אֶרֶץ הָאָבוֹת, "לְרַצּוֹת אֶת אֲבָנֶיהָ, לְחוֹנֵן אֶת עֲפָרָהּ, לְשׁוֹטֵט בַּמְּקוֹמוֹת אֲשֶׁר נִגְלָה אֱלֹקִים לַחוֹזִים וְלַצִּירִים, וּלְהִשְׁתַּטֵּחַ עַל מִבְחַר קְבָרֶיהָ" – כִּלְשׁוֹן הַמְּשׁוֹרֵר רַבִּי יְהוּדָה הַלֵּוִי – "וּכְמוֹ נָתַן אֲרוּכָה וּמַרְפֵּא לְחֳלִי אַהֲבָתוֹ הַסּוֹעֶרֶת לָאָרֶץ הָאֲהוּבָה". אֶרֶץ שֶׁתְּשׁוּקָתוֹ הָרוֹגֶשֶׁת אֵלֶיהָ הִתְמַלְּאָה, סוֹף־סוֹף, בְּעַצְמוֹ.

אֲבָל כָּל מַסָּע נִגְמָר, וְעִם סוֹפוֹ בָּאָה הַפְּרֵדָה. שַׁלְהֵי דְּקַיְטָא תרכ"ט. קָשָׁה עָלָיו הַפְּרֵדָה מִקְּדֻשַּׁת הָאָרֶץ, מֵאֲוִירָהּ הַמַּחְכִּים. הוּא, שֶׁאַהֲבָתוֹ לְאֶרֶץ יִשְׂרָאֵל פָּעֲמָה בּוֹ בְּ"מִדָּה גְּדוּשָׁה מְאֹד", עוֹמֵד לְהִפָּרֵד מֵהָאָרֶץ הָאֲהוּבָה – "אֶרֶץ זָבַת חָלָב וּדְבָשׁ, אֶרֶץ אֲשֶׁר ה' אֱלֹקֶיךָ דּוֹרֵשׁ אוֹתָהּ תָּמִיד". עוֹד מְעַט יֶחְדַּל מִלְּהִתְהַלֵּךְ בְּאֶרֶץ הַצְּבִי, וְשׁוּב יִדְרְכוּ כַּפּוֹת רַגְלָיו עַל הַחוֹל הַחֻלִּין שֶׁל בָּבֶל.

כְּבָר עַכְשָׁו, בְּעוֹדוֹ בְּאֶרֶץ יִשְׂרָאֵל, יָדוֹעַ יָדַע הַבֶּן אִישׁ חַי כִּי עָתִיד הוּא, עַל אַדְמַת הַגּוֹלָה, לִהְיוֹת חוֹלֶה חֳלִי אַהֲבָה וּלְהִתְאַוּוֹת תַּאֲוָה לְאַבְנֶיהָ וְלַעֲפָרָהּ שֶׁל הָאָרֶץ הַקְּדוֹשָׁה. לְפִיכָךְ הִקְדִּים תְּרוּפָה לַמַּכָּה. בְּטֶרֶם צֵאתוֹ, נָטַל עִמּוֹ מֵאַבְנֵי הַמָּקוֹם וְהֶעֱמִיס בְּאַמְתַּחְתּוֹ שַׂקֵּי חוֹל שֶׁנִּלְקְטוּ מֵאַדְמַת הַקֹּדֶשׁ, לְקַיֵּם אֶת מָה שֶׁאָמַר דָּוִד הַמֶּלֶךְ, נְעִים זְמִירוֹת יִשְׂרָאֵל: "כִּי רָצוּ עֲבָדֶיךָ אֶת אֲבָנֶיהָ וְאֶת עֲפָרָהּ יְחֹנֵנוּ".

אֶת הָאֶבֶן הָאֶרֶץ־יִשְׂרְאֵלִית קָבַע הַבֶּן אִישׁ חַי בִּכְנִיסַת בֵּית הַכְּנֶסֶת הַגָּדוֹל בְּבָבֶל, וְאֶת הַחוֹל פִּזֵּר עַל אַדְמַת בֵּית הַכְּנֶסֶת עַצְמוֹ – אֲשֶׁר מֵחֲמַת הַתְּנָאִים הַגֵּיאוֹלוֹגִיִּים שֶׁשָּׂרְרוּ בַּמָּקוֹם, לֹא הָיָה מְרֻצָּף. וְעוֹד חִבֵּר שִׁיר עֶרְגָּה, שֶׁכָּל מִלָּה בּוֹ נוֹשֶׁקֶת חֹלִי:

אֶל הָאָרֶץ חֶמְדָּה טוֹבָה, אַךְ אַךְ אַךְ מִי יַעֲלֵנִי

הרוצה לעלות רגל אחת בגן עדן

הָאַדְמוֹ"ר מִפִּיאָסֶצְנָה, בַּעַל ה'אֵשׁ קֹדֶשׁ' הי"ד, מִתְגַּעְגֵּעַ כָּל יָמָיו לְאֶרֶץ יִשְׂרָאֵל בְּעֹצֶם שֶׁלֹּא יְתֹאַר – "לֹא פַּעַם עָמַד כְּבָר לְחַבֵּר אֶת הַמַּעֲשֶׂה בְּגוּפוֹ וְלַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל, לְכוֹנֵן עָפְרָהּ בְּרִנָּה, אֶלָּא שֶׁהִתְקַשְּׁרוּתוֹ לְצֹאן מַרְעִיתוֹ הִכְרִיעָה אֶת הַכַּף וְהוּא נִשְׁאָר". רַב שֶׁמְּחוֹיָב לְעֲדָתוֹ, נִלְחָם כָּל חַיָּיו בְּמַאֲוַיֵּי הַנֶּפֶשׁ הָעֲמֻקִּים בְּיוֹתֵר שֶׁלּוֹ.

בִּשְׁנַת תר"פ, כְּשֶׁאָחִיו רַבִּי יְשַׁעְיָ'לֶה בָּא מִירוּשָׁלַיִם לְפוֹלִין עַל מְנָת לְהַעֲלוֹת אֶת אִשְׁתּוֹ וּבָנָיו לְאֶרֶץ הַקֹּדֶשׁ, קוֹרֶה מַשֶּׁהוּ פִּתְאוֹמִי לְגַמְרֵי. רַבֵּנוּ, בְּגֹדֶל תְּשׁוּקָתוֹ הָעַזָּה לְקַיֵּם מִצְוַת יִשׁוּב אֶרֶץ יִשְׂרָאֵל, נֶחְפָּז וּמוֹכֵר אֶת כָּל רְכוּשׁוֹ – הַבַּיִת וְכָל דִּבְרֵי הָעֵרֶךְ, כֶּסֶף וְזָהָב שֶׁהָיוּ לוֹ בְּפִיאָסֶצְנָה – וְקוֹנֶה לוֹ אֲחֻזַּת נַחֲלָה בְּפַרְבְּרֵי יְרוּשָׁלַיִם עִיר הַקֹּדֶשׁ.

וְכַאֲשֶׁר הִצִּיעוּ אַנְשֵׁי שְׁלוֹמוֹ לַעֲזֹר לוֹ בְּכַסְפָּם לְבִצּוּעַ הַקְּנִיָּה, אָמַר לָהֶם מָרָן זְצ"ל: "אַ קוּנְץ צוּ קוֹפְּיֶען מִיט יֶעֶנֶעמְס גֶעלְד" (כְּלוּם חָכְמָה הִיא לִקְנוֹת רַק בְּכֶסֶף שֶׁל אֲחֵרִים)? הוּא רוֹצֶה שֶׁזֶּה יִהְיֶה שֶׁלּוֹ – מַמָּשׁ, מִיָּדוֹ שֶׁלּוֹ.

יֵשׁ לְךָ אָדָם הַיּוֹשֵׁב בְּאֶרֶץ יִשְׂרָאֵל בְּגוּפוֹ וְלֹא בְּנַפְשׁוֹ, וְיֵשׁ שֶׁזּוֹכֶה לְקַיְּמָהּ רַק בְּנַפְשׁוֹ וְלֹא בְּגוּפוֹ. אִם לֹא זָכָה הָרַבִּי ז"ל וַעֲדַיִן לֹא אֻכְשַׁר דָּרָא שֶׁיְּקַיֵּם מִצְוַת יְשִׁיבַת אֶרֶץ הַקֹּדֶשׁ בְּגוּפוֹ, הֲרֵי שֶׁבְּנַפְשׁוֹ וְרוּחוֹ הָיָה בָּהּ תָּמִיד, וְכָל יָמָיו חַי בְּיִסּוּרֵי אַהֲבָה לְאֶרֶץ הַקֹּדֶשׁ.

שָׁנִים אֲחָדוֹת אַחֲרֵי כֵן, בִּשְׁנַת ה'תרפ"ג, כְּשֶׁהַמַּצָּב הַכַּלְכָּלִי בְּאֵירוֹפָּה קָשֶׁה, מַצִּיעַ הָרַב יְשַׁעְיָה הַצָּעָה שֶׁנִּרְאֵית הֶגְיוֹנִית לְגַמְרֵי: לִמְכֹּר חֵלֶק מִן הָאֲדָמוֹת שֶׁקָּנָה. הַתְּגוּבָה שֶׁחוֹזֶרֶת אֵלָיו בְּמִכְתָּב אֵינָהּ הֶסְכֵּם רָגִיל – זוֹ זְעָקָה: "דַּע נָא יְקִירִי, כִּי גֶּחָלִים אַתָּה חוֹתֶה עַל רֹאשִׁי, וַהֲרֵי זֶה דּוֹמֶה לְמִי שֶׁעוֹמֵד רֶגֶל אַחַת בְּגַן עֵדֶן, וּבָא הַמַּלְאָךְ... וּלְקָחוֹ בְּיָדוֹ וּבְרַגְלָיו וּמַשְׁלִיכוֹ מִגַּן עֵדֶן לְגֵיהִנֹּם, וּמְנַחֲמוֹ שֶׁבְּגֵיהִנֹּם תִּהְיֶה לוֹ דִּירָה חַמָּה. כֵּן בֶּאֱמֶת הַדָּבָר הַזֶּה. עֵינַי זָלְגוּ דְּמָעוֹת בְּהַעֲלוֹת זֹאת עַל רַעְיוֹנִי, לְהִתְרַחֵק חַס וְשָׁלוֹם עַתָּה, אַחַר שֶׁהָיְתָה לִי קְצָת נְגִיעָה בְּאֶרֶץ יִשְׂרָאֵל".

מְשַׁל שֶׁל רֶגֶל אַחַת בְּגַן עֵדֶן, שֶׁנֶּאֱמַר עַל חֶלְקַת אֲדָמָה קְטַנָּה בְּפַרְבְּרֵי יְרוּשָׁלַיִם. עֵינַיִם שֶׁזָּלְגוּ דִּמְעָה עַל עֶצֶם הָרַעְיוֹן שֶׁל וִתּוּר עַל נְגִיעָה בְּאֶרֶץ יִשְׂרָאֵל.

רגע אחרון — עלייתו האחרונה של הרב מלבורן

הָרוֹפֵא פּוֹשֵׁט אֶת יָדָיו לְצַד, מַפְנֶה אֶת אֶצְבָּעוֹ כְּלַפֵּי הַשָּׁמַיִם. תְּנוּעָה שֶׁאוֹמֶרֶת יוֹתֵר מִכָּל מִלָּה: רַק הַקָּדוֹשׁ בָּרוּךְ הוּא יָכוֹל לַעֲזֹר לְךָ עַכְשָׁו. הַגָּאוֹן רַבִּי אַבְרָהָם צְבִי הַלֵּוִי בֶּעק זְצוּק"ל, רַבָּהּ שֶׁל מֶלְבּוֹרְן שֶׁבְּאוֹסְטְרַלְיָה, שׁוֹמֵעַ אֶת פְּסַק הַדִּין הַזֶּה בְּחַדְרוֹ שֶׁל הָרוֹפֵא בְּבֵית הַחוֹלִים 'אַלְפְרֶד', בְּאִסְרוּ חַג שֶׁל חַג הַשָּׁבוּעוֹת הָאַחֲרוֹן שֶׁל חַיָּיו. הָרוֹפֵא הַיְּהוּדִי, שֶׁלְּצַעֲרֵנוּ אֵינוֹ עֲדַיִן שׁוֹמֵר תּוֹרָה וּמִצְווֹת, מֵבִין שֶׁאֵין בְּיָדוֹ עוֹד מָה לְהָבִיא.

הָרַב, כְּעֵדוּת מְשַׁמְּשׁוֹ הָרַב פִּיקְסְלֶר, עוֹצֵם אֶת עֵינָיו לְדַקָּה אֲרֻכָּה. וְאָז מִסְתּוֹבֵב, וְאוֹמֵר בְּקוֹל בּוֹטֵחַ, קוֹל שֶׁלֹּא מְתְאִים לְאָדָם שֶׁזֶּה עַתָּה שָׁמַע כָּזֹאת בְּשׂוֹרָה: "אֵין דָּבָר הָעוֹמֵד בִּפְנֵי הָרָצוֹן – תָּמִיד רָצִיתִי לַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל". וּמְבַקֵּשׁ שֶׁיִּשְׁאַל אֶת הָרוֹפֵא אִם זֶה בִּכְלָל אֶפְשָׁרִי מִבְּחִינָה רְפוּאִית.

הַתְּשׁוּבָה מַגִּיעָה, וְקוֹצֶבֶת אֶת הַזְּמַן: "אִם הָרַב רוֹצֶה לָטוּס וְלַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל, זֶה צָרִיךְ לִהְיוֹת מִיָּדִי, מַקְסִימוּם תּוֹךְ שְׁבוּעַיִם". וּמֵאוֹתוֹ רֶגַע – מֵרוֹצֶה נֶגֶד הַזְּמַן. הָרַב, "כְּגֶנֵרָל הָעוֹמֵד בִּפְנֵי הַקְּרָב", מַתְחִיל 'לִירוֹת' פְּקֻדּוֹת בְּדֵעָה זַכָּה וּצְלוּלָה: רֹאשִׁית, לְהִפָּרֵד מִן הַקְּהִלָּה. וְקֹדֶם לְכָל דָּבָר, לְהַכְנִיס אֵלָיו מִיָּדִית אֶת בְּנֵי מִשְׁפַּחְתּוֹ – חֲתָנוֹ הַדַּיָּן הָרַב יְהוּדָה הַכֹּהֵן כָּץ, בִּתּוֹ וּנְכָדָיו – לְשַׁתֵּף אוֹתָם בַּסְּקִירָה עַל מַצָּבוֹ.

בְּסִיּוּם דְּבָרָיו, "עֵין לֹא נוֹתְרָה יְבֵשָׁה נֹכַח דְּבָרָיו הַיּוֹצְאִים מִן הַלֵּב". הָרַב מְשַׁתֵּף אֶת בְּנֵי מִשְׁפַּחְתּוֹ בְּהַחְלָטָתוֹ הַסּוֹפִית לַעֲלוֹת לְאֶרֶץ הַקֹּדֶשׁ, וּמְבַקֵּשׁ מֵהֶם לְהִתְגּוֹרֵר אֶצְלוֹ בַּתְּקוּפָה הָרִאשׁוֹנָה. אַחַר כָּךְ הוּא מִתְאוֹשֵׁשׁ הָרִאשׁוֹן מֵהָאֲוִירָה הַטְּעוּנָה, וְנֶחְפָּז לָצֵאת לְמִנְחָה בְּבֵית הַכְּנֶסֶת. "אֲנִי רוֹצֶה לִכְתֹּב מִכְתַּב פְּרֵדָה לְחַבְרֵי הַקְּהִלָּה", הוּא אוֹמֵר. לְמָחֳרָת, יוֹם רְבִיעִי ט' סִיוָן, יוֹשֵׁב הָרַב כִּמְעַט כָּל הַיּוֹם וְכוֹתֵב אֶת מִכְתָּבוֹ, כָּל מִלָּה מְדֻדָּה בְּאֶצְבַּע, הַגָּהָה אַחַר הַגָּהָה, עַד שֶׁמּוֹסֵר אֶת הַמִּכְתָּב עִם חֲתִימָתוֹ. וְהַשְּׁמוּעָה פּוֹשֶׁטֶת כְּאֵשׁ בִּשְׂדֵה קוֹצִים בְּקֶרֶב כָּל בְּנֵי הַקְּהִלָּה, שְׁלוֹשִׁים שָׁנָה שֶׁל רַב וְקְהִלָּה נִפְרָדוֹת.

הָרַב, שֶׁבִּקֵּשׁ לְהִפָּרֵד אִישִׁית מִכָּל אֶחָד וְאֶחָד, פּוֹתֵחַ אֶת דֶּלֶת בֵּיתוֹ מִדֵּי יוֹם, בֵּין הַשָּׁעָה חָמֵשׁ לְשֶׁבַע וְחֵצִי בָּעֶרֶב. מִשְׁפָּחוֹת עַל נָשֵׁיהֶן וְטַפָּן נִכְנָסוֹת בָּזוֹ אַחַר זוֹ, מִי בְּבַקָּשַׁת בְּרָכָה וּמִי בְּבַקָּשַׁת מְחִילָה. דִּמְעוֹת צַעַר וָרֶגֶשׁ מִתְעָרְבוֹת בְּעִרְבּוּבְיָא. הַמִּשְׁפָּחָה מַגִּיעָה לְבַסּוֹף לְאֶרֶץ יִשְׂרָאֵל, וְאַף שֶׁמַּצָּבוֹ הוֹלֵךְ וְמַחְמִיר, מִתְאַמֵּץ בְּכָל כֹּחוֹ לָצֵאת לַתְּפִלּוֹת.

וּבְיוֹם אֶחָד, סְבִיב מִטָּתוֹ, נֶאֱסָפִים כְּאַרְבָּעָה מִנְיָנִים שֶׁל בְּנֵי מִשְׁפַּחְתּוֹ וְתַלְמִידָיו. וְשָׁם, בְּאֶרֶץ יִשְׂרָאֵל שֶׁכָּל יְמֵי חַיָּיו הִשְׁתּוֹקֵק אֵלֶיהָ, הֶחֱזִיר אֶת נִשְׁמָתוֹ הַטְּהוֹרָה לְיוֹצְרָהּ – "בְּ'אֶחָד' מַמָּשׁ, בְּמִיתַת נְשִׁיקָה כִּגְדוֹלֵי הַצַּדִּיקִים, בְּלִי צַעַר בַּגּוּף וּבְלִי טִלְטוּלִים וּזְעָזוּעִים". שְׁבוּעַיִם שֶׁל מֵרוֹץ נֶגֶד הַזְּמַן, כְּדֵי לָמוּת בָּאָרֶץ שֶׁתָּמִיד רָצָה לִחְיוֹת בָּהּ.

כָּל יָמָיו הִתְגַּעְגֵּעַ הָאַדְמוֹ"ר מִפִּיטְסְבּוּרְג זְצ"ל (בַּעַל ה'אֱמוּנַת אַבְרָהָם') לַעֲלוֹת לְאֶרֶץ הַקֹּדֶשׁ. וּכְמוֹ שֶׁכָּתַב בְּעַצְמוֹ: "כַּמָּה דְּמָעוֹת שָׁפַכְתִּי כְּבָר לִפְנֵי הַבּוֹרֵא יִתְבָּרֵךְ שְׁמוֹ, שֶׁלֹּא אֶהְיֶה אֲנִי וּבְנֵי בֵּיתִי וְיוֹצְאֵי חֲלָצַי, מְגֹרָשִׁים חַס וְשָׁלוֹם מִלְּהִסְתּוֹפֵף בְּנַחֲלַת הַשֵּׁם יִתְבָּרֵךְ שְׁמוֹ...".

וְכַאֲשֶׁר זָכָה סוֹף־סוֹף לַעֲלוֹת, וְהִגִּיעַ לָרִאשׁוֹנָה לְאַחַת הֶעָרִים בְּרֶכֶב מְקוֹרָבוֹ, קוֹרֶה מַשֶּׁהוּ בִּלְתִּי צָפוּי לְגַמְרֵי. הוּא מְבַקֵּשׁ פִּתְאוֹם: "הָבָה נַעֲצֹר אֶת הָרֶכֶב וְנֵרֵד". הַנֶּהָג מְבֻלְבָּל: "לְשֵׁם מָה?" וְהָרַבִּי מֵשִׁיב בְּמִלִּים שֶׁלֹּא מַשְׁאִירוֹת מָקוֹם לְוִכּוּחַ: "לִבִּי עוֹלֶה עַל גְּדוֹתָיו. מֻכְרָח אֲנִי לָצֵאת בְּמָחוֹל מֵרֹב שִׂמְחָה!" "לֹא יָכֹלְתִּי לְסָרֵב", מְעִיד הַנֶּהָג. "עָצַרְתִּי בְּצִדֵּי הַדֶּרֶךְ, הָאַדְמוֹ"ר יָצָא עִמִּי בְּמָחוֹל תַּחַת כִּפַּת הַשָּׁמַיִם. הָעוֹבְרִים וְשָׁבִים הִבִּיטוּ בָּנוּ בִּתְדַהֵמָה".

וּמֵאוֹתוֹ רֶגַע נוֹלַד מִנְהָג קָבוּעַ: בְּכָל פַּעַם שֶׁהָיָה יוֹצֵא מֵאֶרֶץ יִשְׂרָאֵל לְחוּץ לָאָרֶץ, כְּשֶׁהָיָה מַגִּיעַ לְיַד כֶּבֶשׁ הַמָּטוֹס, הָיָה נוֹפֵל וּמִשְׁתַּטֵּחַ מְלוֹא קוֹמָתוֹ אַרְצָה, מִתְרַפֵּק עַל אַדְמַת הַקֹּדֶשׁ, מְנַשְּׁקָהּ, וְנִפְרָד מִמֶּנָּה בְּרֹב חִבָּה וּבִדְמָעוֹת שֶׁל שִׁלֵּישׁ. וְכָךְ הָיָה הַדָּבָר חוֹזֵר וְנִשְׁנֶה כְּשֶׁחָזַר מֵחוּץ לָאָרֶץ לְאֶרֶץ יִשְׂרָאֵל.

זמן שמחתנו

מִנְּעוּרָיו הָיְתָה אַהֲבָתוֹ שֶׁל הָאַדְמוֹ"ר מִסָּדִיגוֹרָה, רַבִּי אַבְרָהָם יַעֲקֹב זי"ע בַּעַל ה'אֲבִיר יַעֲקֹב', נִינוֹ שֶׁל רַבִּי יִשְׂרָאֵל מֵרוֹז'ין, לְאֶרֶץ יִשְׂרָאֵל רַבָּה וְעֲצוּמָה. עִנְיָן זֶה הָיָה קָבוּעַ בְּתָכְנִיתוֹ עוֹד מִזְּמַן שֶׁנִּמְלַט מִסָּדִיגוֹרָה לְוִינָה בִּימֵי מִלְחֶמֶת הָעוֹלָם הָרִאשׁוֹנָה.

הַרְבֵּה טִרְחָה וּמַאֲמַצִּים הִשְׁקִיעוּ הַחֲסִידִים לְהוֹצִיאוֹ מִן הַתֹּפֶת הַנָּאצִית בְּוִינָה. מִתְּחִלָּה רָצוּ חֲסִידָיו בְּאָמֶרִיקָה לְסַדֵּר לוֹ וִיזָה לְשָׁם, אוּלָם הוּא סֵרֵב בְּכָל תֹּקֶף לִנְסֹעַ לְכָל מָקוֹם אַחֵר מִלְּבַד אֶרֶץ יִשְׂרָאֵל. גַּם כַּאֲשֶׁר הַקּוֹנְסוּל הָאָמֶרִיקָאִי טִלְפֵּן אֵלָיו וּבִקְּשׁוֹ לִנְסֹעַ לְאָמֶרִיקָה, וּמִשָּׁם יוּכַל לִנְסֹעַ לְאֶרֶץ יִשְׂרָאֵל אִם יִרְצֶה בְּכָךְ, עָמַד עַל דַּעְתּוֹ. לְאַחַר מַאֲמַצִּים קָשִׁים וּמְיַגְּעִים, סֻדְּרָה לוֹ הַוִּיזָה לְאֶרֶץ יִשְׂרָאֵל.

בִּהְיוֹתוֹ סוֹף־סוֹף בְּאֶרֶץ יִשְׂרָאֵל, שָׂמַח שִׂמְחָה גְּדוֹלָה, וּלְהִתְרַגְּשׁוּתוֹ לֹא הָיָה קֵץ וּגְבוּל. בְּרֹאשׁ הַשָּׁנָה הָרִאשׁוֹן לַעֲלִיָּתוֹ טָעָה בַּקִּדּוּשׁ וְאָמַר "זְמַן שִׂמְחָתֵנוּ" בִּמְקוֹם הַנֻּסָּח הַנָּכוֹן. כְּשֶׁנִּשְׁאַל עַל טָעוּתוֹ, הֵשִׁיב: "בֶּאֱמֶת אֵין זוֹ טָעוּת גְּדוֹלָה כָּל כָּךְ, וְכִי יֵשׁ שִׂמְחָה גְּדוֹלָה מִלָּדוּר בְּאֶרֶץ יִשְׂרָאֵל?"

כָּל רֶגַע מִיְּשִׁיבָתוֹ בָּאָרֶץ הָיָה לוֹ לְמֵשִׁיב נֶפֶשׁ. תָּדִיר הֶעֱמִיק לַחְשֹׁב עַל הַזְּכִיָּה הַגְּדוֹלָה שֶׁזִּכָּהוּ הַקָּדוֹשׁ בָּרוּךְ הוּא לְהִנָּצֵל מֵחֲמַת הַמֵּצִיק הַנָּאצִי וְלַעֲלוֹת לְאֶרֶץ יִשְׂרָאֵל, בְּסָרְבוֹ בְּאוֹתוֹ זְמַן לְכָל הַהַצָּעוֹת שֶׁהִצִּיעוּ לוֹ לִנְסֹעַ לְאָמֶרִיקָה. יְשִׁיבָתוֹ בָּאָרֶץ תָּרְמָה לוֹ כָּל הַזְּמַן לְהִתְרוֹמְמוּת נַפְשִׁית גְּדוֹלָה, עַד כְּדֵי כָּךְ שֶׁלֹּא הָיָה מוּכָן בְּשׁוּם אֹפֶן לַעֲזֹב אֶת הָאָרֶץ אֲפִלּוּ כְּדֵי לְבַקֵּר אֶת חֲסִידָיו בְּחוּץ לָאָרֶץ.

אַדְמוֹ"ר אַחֵר הִתְאוֹנֵן פַּעַם בְּפָנָיו כִּי זֶה לוֹ הַבִּקּוּר הַשְּׁלִישִׁי בְּאֶרֶץ יִשְׂרָאֵל, אוּלָם אֵין הוּא מַרְגִּישׁ כָּל כָּךְ אוֹתָהּ הָהֶאָרָה שֶׁהָיְתָה לוֹ בְּבִקּוּרוֹ הָרִאשׁוֹן. וְעַל זֶה הֵשִׁיב לוֹ רַבֵּנוּ הָ'אֲבִיר יַעֲקֹב' זי"ע: "לָנוּ יֵשׁ בְּכָל רֶגַע וְרֶגַע שֶׁל שְׁהוּתֵנוּ בְּאֶרֶץ יִשְׂרָאֵל אוֹתָהּ הָהֶאָרָה וְאוֹתָהּ הַהִשְׁתּוֹקְקוּת כְּפִי שֶׁהָיְתָה לָנוּ בָּרֶגַע הָרִאשׁוֹן שֶׁרַגְלֵנוּ דָּרְכוּ עַל אַדְמַת אֶרֶץ הָאָבוֹת".

היום נתונה לך — הביכורים של הפוניבז'ר רב

קַיִץ תשכ"א, קִבּוּץ שַׁעֲלָבִים בְּעֵמֶק אַיָּלוֹן, מַעֲמָד חֲגִיגִי בְּהִשְׁתַּתְּפוּת נְצִיגֵי הַשִּׁלְטוֹנוֹת. וְהַגָּאוֹן רַבִּי יוֹסֵף שְׁלֹמֹה כַּהֲנֶמַן זְצ"ל, הָרַב מִפּוֹנִיבֶז', קָם וְדִבֵּר בְּהִתְלַהֲבוּת מִשִּׁבְחֵי אַרְצֵנוּ הַקְּדוֹשָׁה. הוּא פּוֹתֵחַ בְּפָסוּק מִפָּרָשַׁת הַשָּׁבוּעַ, פָּרָשַׁת תָּבוֹא, הַשָּׂם דִּבְרֵי מִקְרָא בְּפִי הָאִישׁ מֵבִיא הַבִּכּוּרִים: "הִגַּדְתִּי הַיּוֹם לַה' אֱלֹקֶיךָ כִּי בָאתִי אֶל הָאָרֶץ".

וְשׁוֹאֵל כְּמִשְׁתּוֹמֵם: "וְכִי רַק הָעוֹלִים הַחֲדָשִׁים נִצְטַוּוּ לְהָבִיא בִּכּוּרִים, וְכָל שֶׁהוּא יְלִיד הָאָרֶץ אֵינוֹ בְּמִצְוָה זוֹ?!" אֶלָּא הַכַּוָּנָה, מוֹסִיף מִיָּד, שֶׁגַּם "כָּל אֶחָד מִוָּתִיקֵי אֶרֶץ יִשְׂרָאֵל יֹאהַב אוֹתָהּ אַהֲבַת נְעוּרִים נִצְחִית, כְּאִלּוּ רַק הַיּוֹם בָּא אֵלֶיהָ..." (וְכֵן כָּתַב רַשִׁ"י שְׁמוֹת יג, יא).

אַגַּב: בְּאוֹתוֹ מַעֲמָד הִקְשִׁיב לִדְבָרָיו אֶחָד מֵרָאשֵׁי הַצִּיּוֹנִים הַחִלּוֹנִיִּים, שָׂר בַּמֶּמְשָׁלָה שֶׁיָּשַׁב בֵּין הַקְּרוּאִים, וְהִפְלִיט בְּהִתְפַּעֲלוּת: "מִי מֵאִתָּנוּ אוֹהֵב כָּמוֹהוּ אֶת אֶרֶץ יִשְׂרָאֵל בְּעֻצְמָה כָּזֹאת..." אָדָם חִלּוֹנִי, שֶׁלֹּא בְּקֵלוּת מִתְפַּעֵל מִדִּבְרֵי רַב חָרֵדִי, נִדְהָם מֵעֹצֶם הָאַהֲבָה.

וְאוֹתָהּ תוֹבָנָה בְּדִיּוּק חָזְרָה בְּפִי הַ'נְּתִיבוֹת שָׁלוֹם' מִסְּלוֹנִים זְצ"ל, כְּשֶׁמָּלְאוּ חֲמִשִּׁים שָׁנָה מִיּוֹם עֲלִיָּתוֹ אַרְצָה. הוּא עָרַךְ מְסִבָּה מְיֻחֶדֶת, כְּדֵי לְבָרֵךְ וּלְהוֹדוֹת לַה' בְּמַקְהֵלוֹת עַל כָּל חֲסָדָיו שֶׁזִּכָּהוּ לָשֶׁבֶת בָּאָרֶץ הַזֹּאת, וְאָמַר: "בִּקְרִיאַת בִּכּוּרִים נֶאֱמַר הִגַּדְתִּי 'הַיּוֹם' כִּי בָאתִי אֶל הָאָרֶץ. הַכַּוָּנָה בָּזֶה שֶׁמִּצְוַת יְשִׁיבַת אֶרֶץ יִשְׂרָאֵל צְרִיכָה לִהְיוֹת אֵצֶל יְהוּדִי בִּבְחִינַת בְּכָל יוֹם יִהְיוּ בְעֵינֶיךָ כַּחֲדָשִׁים".

אני נמצא בארץ ישראל

יָמָיו הָאַחֲרוֹנִים שֶׁל הַמַּשְׁגִּיחַ הַגֶּה"צ רַבִּי דֹּב יָפֶה זְצ"ל. חֻלְשָׁה גְּדוֹלָה. הַגּוּף כּוֹשֵׁל. וְאָז, בְּלִי הַתְרָאָה, קוֹרֶה דָּבָר יָפֶה, שֶׁמַּעֵיד עַל דָּבָר עָמֹק יוֹתֵר מִבַּעֲיוֹת הַגּוּף: הוּא מִתְמַלֵּא לְפֶתַע שִׂמְחָה גְּדוֹלָה, סוֹפֵק אֶת יָדָיו בְּהִתְרַגְּשׁוּת שֶׁלֹּא מַתְאִימָה לְחֻלְשָׁתוֹ, וְאוֹמֵר בְּקוֹל שֶׁמַּפְתִּיעַ אֶת הַסּוֹבְבִים אוֹתוֹ:

"אֲנִי נִמְצָא בְּאֶרֶץ יִשְׂרָאֵל, אֲנִי נִמְצָא בְּאֶרֶץ הַקֹּדֶשׁ – כָּל רֶגַע זֶה מִצְוָה"...

לֹא זִכָּרוֹן שֶׁל הֶעָבָר. לֹא תִּקְוָה לֶעָתִיד. אֶלָּא הַכָּרָה בְּמָה שֶׁמַּמָּשׁ קוֹרֶה בָּרֶגַע הַזֶּה, בְּתוֹךְ הַחֻלְשָׁה עַצְמָהּ – שִׂמְחָה שֶׁמְּפַלֶּסֶת לָהּ דֶּרֶךְ אֲפִלּוּ בְּגוּף חַלָּשׁ. אֲדָמַת אֶרֶץ יִשְׂרָאֵל שֶׁמִּתַּחַת לְמִטָּתוֹ הָיְתָה, גַּם בְּרֶגַע חֻלְשָׁה כָּזֶה, מְקוֹר לְשִׂמְחָה.

קניתי רק בזכות האבן עזרא

שֶׁבַע עֶשְׂרֵה שָׁנָה קֹדֶם שֶׁעָלָה בְּעַצְמוֹ לָגוּר בְּאֶרֶץ יִשְׂרָאֵל, עוֹשֶׂה הַגָּאוֹן רַבִּי יְחֶזְקֵאל אַבְרַמְסְקִי זְצ"ל, בַּעַל 'חֲזוֹן יְחֶזְקֵאל', מַעֲשֶׂה מוּזָר. בְּקַיִץ תרצ"ד, בְּעוֹדוֹ בְּחוּצָה לָאָרֶץ, הוּא רוֹכֵשׁ בְּכַסְפּוֹ חֶלְקַת אֲדָמָה קְטַנָּה בְּאֶרֶץ הַגָּלִיל, לִהְיוֹת לוֹ לַאֲחֻזָּה. וְהָיָה מְסַפֵּר לְמְיֻדָּעָיו, כְּשֶׁרוּחוֹ טוֹבָה עָלָיו, שֶׁמֵּעוֹלָם – בְּאֶף אַחַת מֵאוֹתָן שֶׁבַע עֶשְׂרֵה שָׁנִים – לֹא בִּקֵּר בַּמָּקוֹם וְלֹא רָאָה אֶת הַמִּגְרָשׁ בְּעֵינָיו.

וְכַאֲשֶׁר שׁוֹאֲלִים אוֹתוֹ מְיֻדָּעָיו מַדּוּעַ, הַתְּשׁוּבָה שֶׁלּוֹ אֵינָהּ עֵסֶק, וְאֵינָהּ הַשְׁקָעָה כַּלְכָּלִית. הִיא פָּסוּק אֶחָד שֶׁל פָּרְשָׁן מֵאוֹת שָׁנִים קֹדֶם: "קָנִיתִי אֶת חֶלְקַת הָאֲדָמָה אַךְ וְרַק מִפְּנֵי דִּבְרֵי הָאִבֶּן עֶזְרָא" (בְּפָרָשַׁת וַיִּשְׁלַח), שֶׁמְּבָאֵר מַדּוּעַ הִזְכִּיר הַכָּתוּב שֶׁיַּעֲקֹב אָבִינוּ קָנָה חֶלְקַת שָׂדֶה: "לְהוֹדִיעַ כִּי מַעֲלָה גְּדוֹלָה יֵשׁ לְאֶרֶץ יִשְׂרָאֵל, מִי שֶׁיֵּשׁ לוֹ חֵלֶק בָּהּ חָשׁוּב הוּא כְּחֵלֶק לָעוֹלָם הַבָּא".

פָּסוּק שֶׁל פָּרְשָׁן, וְעֵסֶק אֲמִתִּי שֶׁל אֲדָמָה שֶׁלֹּא רָאָה. זוֹ הַמִּדָּה שֶׁל אָדָם שֶׁמַּאֲמִין בְּמָה שֶׁקָּרָא – עַד כְּדֵי כָּךְ שֶׁהוּא מוֹצִיא כֶּסֶף עַל אֲדָמָה שֶׁמֵּעוֹלָם לֹא רָאָה, וְשׁוֹמֵר עַל אוֹתָהּ אֲמוּנָה שֶׁבַע עֶשְׂרֵה שָׁנָה תְּמִימוֹת.

כל אדמה למטרה רוחנית — קנייה בכסף מלא

בְּבִקּוּרוֹ הָרִאשׁוֹן בְּאַרְצֵנוּ הַקְּדוֹשָׁה, תָּר הָאַדְמוֹ"ר מִצַּאנְז בַּעַל ה'שְׁפַע חַיִּים' אֶת הָאָרֶץ לְאָרְכָּהּ וּלְרָחְבָּהּ, כְּשֶׁהוּא מְחַפֵּשׂ בְּעֵינָיו הַקְּדוֹשׁוֹת אַחֲרֵי מָקוֹם מַתְאִים וְרַב־יְדַיִם, בּוֹ יוּכַל לְהִשְׁתַּקֵּעַ וּלְכוֹנֵן אֶת קִרְיַת צַאנְז כְּפִי שֶׁרָקַם בַּחֲזוֹנוֹ. פּוֹנִים אֵלָיו מִירוּשָׁלַיִם עִיר הַקֹּדֶשׁ שֶׁיָּבוֹא וְיָכוֹנֵן בָּהּ אֶת קִרְיָתוֹ. הַצָּעָה מְפַתָּה מְאֹד – יְרוּשָׁלַיִם, עִיר הַקֹּדֶשׁ, קְדֻשָּׁה יְתֵרָה.

הוּא דּוֹחֶה. וְהַנִּמּוּק חוֹשֵׂף אֶת גֹּדֶל הָרַאִיָּה שֶׁלּוֹ: "יְרוּשָׁלַיִם יֵשׁ בָּהּ בָּרוּךְ הַשֵּׁם קְדֻשָּׁה יְתֵרָה וְתוֹרָה בְּשֶׁפַע, בְּעוֹד אֲנִי רוֹצֶה לְהוֹסִיף בְּאֶרֶץ יִשְׂרָאֵל עִיר חֲדָשָׁה, אֲשֶׁר עִם הֲקָמָתָהּ תִּגְדַּל הַקְּדֻשָּׁה וְיִתְרַבֶּה אוֹר הַתּוֹרָה". מִטַּעַם זֶה דָּחָה גַּם אֶת הַהַצָּעָה לִבְנוֹת אֶת קִרְיַת צַאנְז בִּבְנֵי בְּרַק.

לְאַחַר שֶׁהֶחְלִיט עַל שֶׁטַח עִיר נְתַנְיָה, יִפָּה אֶת כֹּחַ חֲבֵרֵי הַוַּעַד לִרְכֹּשׁ אֶת הַמִּגְרָשִׁים בִּשְׁמוֹ וּבִשְׁלִיחוּתוֹ – בְּכֶסֶף מָלֵא דַּוְקָא. לְיָמִים, כְּשֶׁהִתְיַחֵס בְּשִׂיחָה לְעִנְיַן רְכִישַׁת הָאֲדָמוֹת שֶׁל קִרְיַת צַאנְז בְּכֶסֶף מָלֵא, הִזְכִּיר הָרַבִּי אֶת דִּבְרֵי רַבֵּנוּ בְּחַיֵּי הַמַּצְבִּיעַ עַל כָּךְ, שֶׁרֹב שִׁטְחֵי אַרְצֵנוּ הַקְּדוֹשָׁה נִכְבְּשׁוּ עַל יְדֵי יִשְׂרָאֵל לְפִי חֶרֶב, אוּלָם אֶת הַשְּׁטָחִים הַמְּעֻלִּים בְּיוֹתֵר קָנוּ בְּכֶסֶף מָלֵא, וְהֵם שְׁלֹשָׁה: מְעָרַת הַמַּכְפֵּלָה, הַר הַמּוֹרִיָּה, וְהַר גְּרִזִּים וְהַר עֵיבָל – הַמָּקוֹם שֶׁקִּבְּלוּ יִשְׂרָאֵל עֲלֵיהֶם אֶת הַתּוֹרָה בִּבְרִית וּבְאָלָה.

"מִכָּאן נִמְצֵאנוּ לְמֵדִים", הוֹסִיף הָרַבִּי, "שֶׁכָּל אֲדָמָה הַמְּיֹעֶדֶת לְמַלֵּא תַּפְקִיד רוּחָנִי בְּחַיֵּי עַמֵּנוּ, צָרִיךְ לִקְנוֹתָהּ בְּכֶסֶף מָלֵא דַּוְקָא, וְלֹא לְקַבֵּל אוֹתָהּ בְּחִנָּם בְּשׁוּם פָּנִים". וּכְבָר הִרְחִיבוּ חֲזַ"ל אֶת הַדִּבּוּר בַּזֹּהַר הַקָּדוֹשׁ עַל אוֹדוֹת חִין עֶרְכָּהּ שֶׁל כָּל מִצְוָה דַּוְקָא כְּשֶׁמְּשַׁלְּמִים בַּעֲדָהּ מְחִיר מָלֵא, וְלֹא כְּשֶׁמְּקַיְּמִין אוֹתָהּ בְּחִנָּם.

דירה בארץ ישראל לא מוכרים

יְהוּדִי רוֹצֶה לַעֲבֹר דִּירָה. וְעוֹלֶה בְּדַעְתּוֹ רַעְיוֹן שֶׁנִּרְאֶה לוֹ חָכָם: לִמְכֹּר אֶת דִּירָתוֹ, לָגוּר בִּשְׂכִירוּת, וּבַכֶּסֶף שֶׁיְּקַבֵּל מִמְּכִירַת הַדִּירָה יוּכַל לָשֶׁבֶת וְלַעֲסֹק בְּתוֹרָה שָׁנִים רַבּוֹת. חֶשְׁבּוֹן פָּשׁוּט, שֶׁנִּרְאֶה נָכוֹן וְנוֹח. הוּא נִגַּשׁ לְמָרָן הַגַּר"ח קָנְיֶבְסְקִי זְצ"ל וְשׁוֹאֵל אוֹתוֹ הַאִם כְּדַאי לַעֲשׂוֹת כָּךְ.

וְהַתְּשׁוּבָה מְנַפֶּצֶת אֶת הַתָּכְנִית בְּמִשְׁפָּט אֶחָד קָצָר: "דִּירָה בְּאֶרֶץ יִשְׂרָאֵל לֹא מוֹכְרִים, וְאִם צָרִיךְ לַעֲבֹר דִּירָה, אָז מוֹכְרִים וְקוֹנִים קָטָן יוֹתֵר, אֲבָל לֹא נִשְׁאָרִים לְלֹא בַּיִת בְּאֶרֶץ יִשְׂרָאֵל. בְּעֶזְרַת הַשֵּׁם תּוּכַל לִלְמֹד תּוֹרָה, מֵהַכֶּסֶף שֶׁיִּשָּׁאֵר לְךָ אַחֲרֵי הַמְּכִירָה".

וְכָךְ הָיָה: הַיְּהוּדִי מָכַר אֶת דִּירָתוֹ, קָנָה דִּירָה גְּדוֹלָה יוֹתֵר בְּאֵזוֹר אַחֵר, וְנִשְׁאַר לוֹ עוֹדֵף כֶּסֶף – עוֹדֵף שֶׁמַּסְפִּיק בְּדִיּוּק, וּבְעֶזְרַת ה' עַד הַיּוֹם הַזֶּה הוּא יוֹשֵׁב וְעוֹסֵק בַּתּוֹרָה. וְכַלָּשׁוֹן הַנּוֹסֶפֶת הַמּוּבֵאת בְּשֵׁם הַחָתָם סוֹפֵר: מִי שֶׁמַּפְסִיד חֵלֶק בְּאֶרֶץ יִשְׂרָאֵל נִקְרָא בְּמֻבָּן מְסֻיָּם "מִיתָה" – כִּי יְדִיעַת הַהֲפָכִים אַחַת, וְהַחַיִּים בְּאֶרֶץ יִשְׂרָאֵל הֵם הַהֶפֶךְ הַגָּמוּר.

סִפֵּר הַמַּשְׁגִּיחַ דְּ'קוֹל תּוֹרָה', הַגֶּה"צ רַבִּי גְּדַלְיָה אַיְזְמַן זְצ"ל: "לִפְנֵי כְּאַרְבָּעִים שָׁנָה רָאִיתִי פַּעַם אֶת רַבִּי יְחֶזְקֵאל אַבְרַמְסְקִי זְצ"ל עוֹמֵד וּמִסְתַּכֵּל מֵחַלּוֹנוֹת הַיְּשִׁיבָה בְּקוֹל תּוֹרָה, לְעֵבֶר הָעֵמֶק שֶׁהָיָה נִשְׁקָף מֵהַיְּשִׁיבָה, עַד קִרְיַת יוֹבֵל – זֹאת עוֹד לִפְנֵי שֶׁבָּנוּ אֶת כָּל הַבָּתִּים". גָּאוֹן זָקֵן, עוֹמֵד לְבַדּוֹ, מַבִּיט בְּעֵמֶק כִּמְעַט רֵיק, שֶׁטֶרֶם נִבְנוּ עָלָיו הַשְּׁכוּנוֹת.

וְאָז – דְּמָעוֹת. "הוּא הֻקְסַם מֵהַנּוֹף עַד כְּדֵי דְּמָעוֹת וְהִפְטִיר", מְסַפֵּר הָרַב אַיְזְמַן: "כָּל שֶׁלֹּא אָמַר אֶרֶץ חֶמְדָּה טוֹבָה וּרְחָבָה, לֹא יָצָא יְדֵי חוֹבָתוֹ" (כִּלְשׁוֹן הַגְּמָרָא בִּבְרָכוֹת מח, ב) – "מִי שֶׁלֹּא מִתְפַּעֵל מֵהַיֹּפִי שֶׁל אֶרֶץ יִשְׂרָאֵל, לֹא יָצָא יְדֵי חוֹבָתוֹ". לֹא בְּרָכָה שֶׁל מִלִּים בִּלְבַד, כְּשֵׁם שֶׁלֹּא מַסְפִּיק לִזְכֹּר אֶת הַנֻּסָּח – בְּרָכָה שֶׁדּוֹרֶשֶׁת עַיִן שֶׁמִּתְפַּעֶלֶת בֶּאֱמֶת.

מה יפית ארצנו הקדושה

עַל הָרַב מִפּוֹנִיבֶז' הַגְּרִי"ש כַּהֲנֶמַן זְצ"ל מְסֻפָּר, שֶׁהָיָה מַרְגֵּשׁ וּמְלַבֵּב לִרְאוֹתוֹ בְּהִתְפַּעֲמוּתוֹ מִיֹּפְיָהּ שֶׁל אֶרֶץ יִשְׂרָאֵל. רַבִּים מִבְּנֵי הַתּוֹרָה עוֹד זוֹכְרִים בְּוַדַּאי אֶת הוֹפָעָתוֹ, בְּטֵבֵת תש"ו, בְּכְפַר חֲסִידִים, הָרוֹבֵץ עַל פְּנֵי אֶחָד הָרְכָסִים שֶׁל הַר הַכַּרְמֶל, לְרַגְלֵי חֲתֻנָּתוֹ שֶׁל הַגָּאוֹן רַבִּי אֵלִיָּהוּ אֱלִיעֶזֶר מִישְׁקוֹבְסְקִי לְרַב הַמָּקוֹם. עֵינָיו חוֹבְקוֹת בְּמַבָּט נִלְהָב אֶת כָּל הַנּוֹף הַמִּשְׁתָּרֵעַ לְרַגְלָיו, וְכָל כֻּלּוֹ מִתְרוֹנֵן מִגִּיל וְאֹשֶׁר. וּפִתְאוֹם, פּוֹרֶצֶת מִפִּיו קְרִיאָה חַמָּה שֶׁלֹּא יָכוֹל הָיָה לְעַצֹּר: "הֶאָח, ווִי שֵׁיין אִיז אֶרֶץ יִשְׂרָאֵל, מַה יָּפִית אַרְצֵנוּ הַקְּדוֹשָׁה!"

וּבְאוֹתוֹ רֶגַע, כָּל הָעָם מִסָּבִיב מִשְׁתָּאֶה אֶל הַמֶּחֱזֶה, וְאִישׁ אֶל רֵעֵהוּ יְלַחֲשׁוּן: הֲרָאִיתָ כָּזֹאת מִיָּמֶיךָ? מִי יַשְׁוֶה לוֹ בְּעֹז אַהֲבָתוֹ לְאַדְמַת קֹדֶשׁ – הֲרֵי כְּמוֹהוּ כְּכִנּוֹר חַי לְשִׁירַת הַפַּיְטָן הָאֱלֹקִי רַבֵּנוּ יְהוּדָה הַלֵּוִי, הַמְּבַטֵּא כִּסּוּפִים דּוֹמִים לְאֶרֶץ הַקֹּדֶשׁ.

וּכְבָר הָיָה מַעֲשֶׂה, שֶׁטִּיֵּל פַּעַם בִּבְנֵי בְּרַק בֵּאֵזוֹר הַפַּרְדֵּסִים אֲשֶׁר מִסָּבִיב לְקִרְיַת הַיְּשִׁיבָה, בְּטֶרֶם הִבָּנוֹתָהּ, וְנִלְוָה אֵלָיו אֶחָד הַסּוֹפְרִים הַחֲשׁוּבִים. לְפֶתַע עָצַר הָרַב מֵהִלּוּכוֹ, וְסָנַט בְּבֶן לְוָיָתוֹ: "סוֹפֵר הִנְּךָ? מְשׁוֹרֵר? בַּעַל נֶפֶשׁ פִּיּוּטִית? אֵיזֶה נוֹף שֶׁל תִּפְאָרָה מִשְׁתָּרֵעַ כָּאן לְעֵינֶיךָ, מַמָּשׁ כְּמוֹ שֶׁכָּתוּב 'יְפֵה נוֹף מְשׂוֹשׂ כָּל הָאָרֶץ'! כֵּיצַד, אֵפוֹא, הִנְּךָ נִשְׁאָר אָדִישׁ לְמַרְאֵהוּ, אִם לֹא שֶׁשּׁוֹכֶנֶת בְּקִרְבְּךָ נְשָׁמָה קְרִירָה..."

הולך באותם הרים שהלך בהם אברהם אבינו

הַמִּשְׁנָה קוֹבַעַת דִּין חָמוּר לְכְאוֹרָה: "הַמַּפְסִיק מִמִּשְׁנָתוֹ וְאוֹמֵר: מַה נָּאֶה אִילָן זֶה, מַה נָּאֶה נִיר זֶה – הֲרֵי זֶה מִתְחַיֵּב בְּנַפְשׁוֹ". הַגָּאוֹן רַבִּי מָרְדְּכַי גִיפְטֶר זְצ"ל, רֹאשׁ יְשִׁיבַת טֶעלְז בְּאַרְצוֹת הַבְּרִית, שִׁתֵּף אֶת הָג"ר יְהוֹשֻׁעַ לִיף, רֹאשׁ יְשִׁיבַת נֵר יַעֲקֹב, בְּחִדּוּשׁ נִפְלָא, שֶׁכָּתַב עָלָיו בְּמִכְתָּב לְהַגְרִי"י רוּדֶרְמַן, רֹאשׁ יְשִׁיבַת נֵר יִשְׂרָאֵל: הוּא הֵבִין אֶת דִּין הַמִּשְׁנָה כָּל חַיָּיו בְּדֶרֶךְ פְּשׁוּטָה.

"עַד שֶׁהִגַּעְתִּי לְאֶרֶץ יִשְׂרָאֵל", הוּא כּוֹתֵב, "הֵבַנְתִּי שֶׁהַכַּוָּנָה הִיא לְאָדָם הַשָּׁקוּעַ בְּ'קְצוֹת', בְּ'נְתִיבוֹת' אוֹ בְּרַבִּי עֲקִיבָא אֵיגֶר, וְעוֹצֵר מִלִּמּוּדוֹ כְּדֵי לְהִתְפַּעֵל מִיֹּפְיוֹ שֶׁל אִילָן. הִתְנַהֲגוּת כָּזוֹ מְעִידָה עַל זִלְזוּל בְּלִמּוּד הַתּוֹרָה, וְלָכֵן הָאָדָם מִתְחַיֵּב בְּנַפְשׁוֹ".

"אֲבָל עַכְשָׁו כְּשֶׁהִגַּעְתִּי לְאֶרֶץ יִשְׂרָאֵל, אֲנִי מוֹצֵא אֶת עַצְמִי מְהַלֵּךְ בְּאוֹתָן דְּרָכִים שֶׁבָּהֶן הִתְהַלֵּךְ אַבְרָהָם אָבִינוּ... אֲנִי, מָרְדְּכַי גִיפְטֶר, הוֹלֵךְ בְּאוֹתָם הָהָרִים שֶׁהָלַךְ בָּהֶם אַבְרָהָם אָבִינוּ. מִי שֶׁמַּחְשִׁיב הַכָּרָה כָּזוֹ – שֶׁל מַרְאֵה אִילָן אוֹ הָר יָפֶה בְּאֶרֶץ יִשְׂרָאֵל – וּמַפְסִיק מִלִּמּוּדוֹ, אֵינוֹ מֵבִין שֶׁגַּם זֶה קֹדֶשׁ – שֶׁגַּם הֶהָרִים הָאֵלֶּה הֵם בִּטּוּי שֶׁל אֱלֹקוּת... הֲרֵי זֶה מִתְחַיֵּב בְּנַפְשׁוֹ".

וְעוֹד הוֹסִיף פְּרָט מְעַנְיֵן: בְּהֶמְשֵׁךְ מָצָא שֶׁבֵּאוּר זֶה מוּבָא גַּם בְּשֵׁם הַבַּעַל שֵׁם טוֹב, וּמְעַנְיֵן שֶׁגַּם הַבַּעַל שֵׁם טוֹב הִשְׁתּוֹקֵק לְהַגִּיעַ לְאֶרֶץ יִשְׂרָאֵל, אַךְ מַאֲמָצָיו לֹא עָלוּ בְּיָדוֹ – סוֹגֵר מַעְגָּל אֶל הַסִּפּוּרִים הָרִאשׁוֹנִים בְּסֵפֶר זֶה.

אני שמח שמקיימים מצות יישוב הארץ

שַׁבָּת בְּבֵית נִכְדּוֹ שֶׁל הַמַּשְׁגִּיחַ הַגֶּה"צ רַבִּי דֹּב יָפֶה זְצ"ל. שַׁבָּת רְגִילָה, לְכְאוֹרָה. אֲבָל הַנֶּכֶד שָׂם לֵב לְמַשֶּׁהוּ: סַבָּא מַבִּיט וּמִתְעַנְיֵן, שׁוּב וָשׁוּב, בְּבִנְיָן שֶׁבּוֹנִים מוּל הַבַּיִת. לֹא מַבָּט חוֹלֵף – עִנְיָן מַמָּשִׁי, הֲנָאָה וְשִׂמְחָה נִכֶּרֶת עַל פָּנָיו.

"שָׁאַלְתִּיו מַדּוּעַ סַבָּא כָּל כָּךְ מִתְעַנְיֵן?" מְסַפֵּר הַנֶּכֶד. וְהַתְּשׁוּבָה, קְצָרָה כְּמוֹ שֶׁהִיא, מְגַלָּה עוֹלָם שָׁלֵם שֶׁל הַשְׁקָפָה: "אֲנִי שָׂמֵחַ שֶׁמְּקַיְּמִים מִצְוַת יִשּׁוּב אֶרֶץ יִשְׂרָאֵל!" לֹא כִּי הוּא בּוֹנֶה. לֹא כִּי הוּא מַרְוִיחַ. שִׂמְחָה נְקִיָּה עַל עֶצֶם הַמִּצְוָה הַנֶּעֱשֵׂית מוּל עֵינָיו, בְּבִנְיָן שֶׁאֵינוֹ נוֹגֵעַ לוֹ אִישִׁית.

מצפה הושעיה — שמחתו של הרב מרדכי אליהו

נָסִיעָה שֶׁל שִׁגְרָה בְּצָפוֹן הָאָרֶץ. סִפֵּר הָרַב שָׁאוּל דִּימְרִי: "פַּעַם נָסַעְתִּי עִם הַגָּאוֹן רַבִּי מָרְדְּכַי אֵלִיָּהוּ זְצ"ל בְּצָפוֹן הָאָרֶץ, וּבְדַרְכֵּנוּ עָבַרְנוּ בְּכַמָּה כְּפָרִים עַרְבִיִּים כְּמוֹ טוּרְעָאן, וְרָאִיתִי שֶׁבְּכָל הַיִּשּׁוּבִים הַלָּלוּ אֵין יְהוּדִים מִבָּנֶיהָ שֶׁל אֶרֶץ יִשְׂרָאֵל". וְלְפֶתַע רָאִינוּ יִשּׁוּב בִּבְנִיָּה, וּלְפִי הַסֵּדֶר וְסִגְנוֹן הַבְּנִיָּה הָיָה נִכָּר שֶׁמְּדֻבָּר בְּיִשּׁוּב יְהוּדִי חָדָשׁ. הָרַב בִּקֵּשׁ לַעֲצֹר אֶת הָרֶכֶב וְשָׁאַל מָה הַיִּשּׁוּב הַזֶּה.

"אָמַרְתִּי לוֹ שֶׁבִּשְׁבִיל לְיַהֵד אֶת הַגָּלִיל הֶחְלִיטוּ לְהָקִים יִשּׁוּבִים קְטַנִּים בִּמְקוֹמוֹת אֶסְטְרָטֶגִיִּים שֶׁנִּקְרָאִים 'מִצְפִּים', בִּכְדֵי לִשְׁמֹר עַל אַדְמוֹת הַגָּלִיל לְמַעַן עַם יִשְׂרָאֵל, וְשֶׁלֹּא יִשְׁתַּלְּטוּ עֲלֵיהֶם עַרְבִים. הִצְבַּעְתִּי עַל הַכֵּלִים הַכְּבֵדִים שֶׁפָּרְצוּ שָׁם אֶת הַדֶּרֶךְ וְעַל הַקַּרְוָנִים שֶׁמֻּצָּבִים שָׁם וְאָמַרְתִּי לוֹ: כָּאן מוּקָם מִצְפֶּה הוֹשַׁעְיָה, כַּנִּרְאֶה עַל שֵׁם רַבִּי הוֹשַׁעְיָה שֶׁיָּשַׁב בַּיִּשּׁוּב צִפּוֹרִי שֶׁנִּמְצָא בְּסָמוּךְ לְכָאן".

הַפָּנִים שֶׁל הָרַב אֵלִיָּהוּ הוֹאֲרוּ מְאֹד, וְנִכָּר בּוֹ שֶׁהוּא שָׂמֵחַ שִׂמְחָה גְּדוֹלָה. הָרַב יָצָא מֵהָרֶכֶב, וְהִתְחִיל לְדַבֵּר עַל חֲשִׁיבוּת הַבִּנְיָן וְהַיִּשּׁוּב שֶׁל אֶרֶץ יִשְׂרָאֵל. הוּא הִזְכִּיר אֶת נְבוּאַת יִרְמְיָהוּ "עוֹד אֶבְנֵךְ וְנִבְנֵית בְּתוּלַת יִשְׂרָאֵל" (יִרְמְיָהוּ לא), הִזְכִּיר אֶת הַנְּבוּאָה בְּסֵפֶר יְחֶזְקֵאל שֶׁאוֹמֶרֶת כִּי כָּל בַּיִת שֶׁנִּבְנֶה בְּאֶרֶץ יִשְׂרָאֵל הוּא סִימָן לְטָהֳרָתָם שֶׁל יִשְׂרָאֵל הַמִּתְחַזֶּקֶת: "בְּיוֹם טַהֲרִי אֶתְכֶם מִכָּל עֲוֹנוֹתֵיכֶם וְהוֹשַׁבְתִּי אֶת הֶעָרִים וְנִבְנוּ הֶחֳרָבוֹת" (יְחֶזְקֵאל לו, לג).

הָרַב שָׂמַח כָּל כָּךְ, עַד שֶׁבִּקֵּשׁ אֶת הַגְּלִימָה שֶׁלּוֹ, לָבַשׁ אוֹתָהּ – שָׁם, בְּצַד הַדֶּרֶךְ – וְהִתְחִיל לָשִׁיר אֶת הַמִּלִּים מִן הַהַלֵּל: "אָנָּא ה' הוֹשִׁיעָה נָּא". מִן הַסְּתָם בִּגְלַל הַשֵּׁם שֶׁל הַיִּשּׁוּב, מִצְפֶּה הוֹשַׁעְיָה.

"הָאֱמֶת הִיא", מְסַיֵּם הָרַב דִּימְרִי, "שֶׁאֲנִי קְצָת הִתְבַּיַּשְׁתִּי בְּעַצְמִי שֶׁאֵינֶנִּי שָׂמֵחַ שִׂמְחָה כָּל כָּךְ גְּדוֹלָה בְּכָל פַּעַם שֶׁאֲנִי עוֹבֵר כָּאן וְרוֹאֶה אֶת בִּנְיַן הָאָרֶץ. הִתְרַגַּלְנוּ, וְלֹא שָׂמִים לֵב לְגֹדֶל הַמַּעֲשֶׂה שֶׁנַּעֲשֶׂה עַל יָדֵינוּ יוֹם יוֹם. טוֹב שֶׁנָּסַעְתִּי עִם הָרַב אֵלִיָּהוּ שֶׁהֶרְאָה לָנוּ כִּי כָּל יִשּׁוּב הוּא גִּלּוּי שְׁכִינָה מַמָּשׁ. שִׂמְחָה עֲנָקִית שֶׁל מַמָּשׁ".

הזוננפלד בין השדות

קַיִץ, שָׁנָתוֹ הָאַחֲרוֹנָה שֶׁל הַגְּרִי"ח זוֹנֶנְפֶלְד. הוּא כְּבָר חַלָּשׁ מְאֹד, וְהַהֲלִיכָה בָּרֶגֶל בִּלְתִּי אֶפְשָׁרִית כְּלָל. אֲבָל הוּא לֹא מְוַתֵּר. הוּא נוֹסֵעַ, לְעִתִּים קְרוֹבוֹת, בְּרֶכֶב, כְּדֵי לִרְאוֹת אֶת הַשְּׁכוּנוֹת הַחֲדָשׁוֹת שֶׁל יְרוּשָׁלַיִם. כִּמְעַט שֶׁלֹּא הִשְׁאִיר שְׁכוּנָה שֶׁלֹּא בִּקֵּר בָּהּ.

כְּשֶׁנּוֹסְדָה הַמּוֹשָׁבָה "מַחֲנֵה יִשְׂרָאֵל" עַל יְדֵי אֲגֻדַּת יִשְׂרָאֵל, נָסַע מָרָן זיע"א לְבַקְּרָהּ וּלַחֲנֹךְ בָּהּ אֶת רַבָּנוּתוֹ שֶׁל אֶחָד מֵחַכְמֵי הַיְּשִׁיבָה שֶׁבִּירוּשָׁלַיִם. וּפִתְאוֹם, רוֹאֶה מַרְאֶה שֶׁמַּדְלִיק בּוֹ אוֹר: יְהוּדִים חֲרֵדִים אוֹחֲזִים בְּמַחֲרֵשָׁה וּמַעְדֵּר, חוֹרְשִׁים אֶת אַדְמַת הַקֹּדֶשׁ. צָהֳלָתוֹ הָיְתָה בְּפָנָיו, עֵינָיו הַמְּאִירוֹת הִבְרִיקוּ שִׁבְעָתַיִם. הוּא לֹא מִסְתַּפֵּק בְּמַבָּט מִן הַצַּד.

פִּתְאוֹם נִגַּשׁ מָרָן זיע"א אֶל אֶחָד הַחוֹרְשִׁים, הוֹצִיא אֶת הַמַּחֲרֵשָׁה מִיָּדָיו, וְנִהֵל בְּעַצְמוֹ אֶת צֶמֶד הַבָּקָר מֶשֶׁךְ מֶטְרִים אֲחָדִים – רַב זָקֵן, פּוֹסֵק הֲלָכָה מֻסְמָךְ, מְנַהֵל בְּעַצְמוֹ צֶמֶד בָּקָר בַּשָּׂדֶה. הוּא נִגַּשׁ גַּם אֶל בְּאֵר הַמַּיִם וּמִלֵּא כַּמָּה כַּדֵּי מַיִם, וּשְׂפָתָיו הִבִּיעוּ: "וּשְׁאַבְתֶּם מַיִם בְּשָׂשׂוֹן". אַחֲרֵי כֵן פָּקַד עַל כָּל הַנּוֹכְחִים לִשְׁאֹב, וְלוֹמַר בִּשְׁעַת מַעֲשֶׂה אֶת ט"ו שִׁיר הַמַּעֲלוֹת שֶׁאָמַר דָּוִד בְּשָׁעָה שֶׁהֶעֱלָה אֶת הַשִּׁיתִין.
